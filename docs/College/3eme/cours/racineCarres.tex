\section*{\color{myred}I. La racine carrée d'un nombre réel positif}

\subsection*{\color{myred} Activité 1}
\begin{enumerate}[label=\textbf{\arabic*.}]
    \item Écrire sous forme d'une puissance les nombres suivants :
    \[
    \mathbf{4 \quad ; \quad 9 \quad ; \quad 25 \quad ; \quad 36 \quad ; \quad 81}
    \]

    \item Compléter le tableau suivant en utilisant la calculatrice :
    
    \vspace{0.3cm}
    \begin{center}
    \renewcommand{\arraystretch}{1.8}
    \begin{tabular}{|c|c|c|c|c|c|}
        \hline
        Le nombre $a$ & 4 & 9 & 25 & 36 & 81 \\ \hline
        $a = \square^2$ & \hspace{1cm} & \hspace{1cm} & \hspace{1cm} & \hspace{1cm} & \hspace{1cm} \\ \hline
        $\sqrt{a}$ & & & & & \\ \hline
    \end{tabular}
    \end{center}
    \vspace{0.3cm}

    \item Que remarquez-vous ?
\end{enumerate}

\subsection*{\color{myred} Définition}
Soit $a$ un nombre réel positif. La \textbf{racine carrée} de $a$, notée $\sqrt{a}$, est le nombre positif dont le carré est $a$.
\subsection*{\color{myred} Exemple 1}
$4$ est le carré de $\mathbf{2}$, donc : $\mathbf{\sqrt{4} = 2}$ \\[0.2cm]
$\mathbf{25}$ est le carré de $\mathbf{5}$, donc : $\mathbf{\sqrt{25} = 5}$ \\[0.2cm]
$\mathbf{\sqrt{64} = 8 \quad ; \quad \sqrt{81} = 9 \quad ; \quad \sqrt{0} = 0 \quad ; \quad \sqrt{1} = 1}$
\subsection*{\color{myred} Propriété}
Pour tout nombre réel positif $a$, on a :
\[
\mathbf{(\sqrt{a})^2 = a \quad ; \quad \sqrt{a^2} = a \quad ; \quad \sqrt{(-a)^2} = a}
\]
\subsection*{\color{myred} Exemple 2}
\[
\mathbf{(\sqrt{13})^2 = 13 \quad ; \quad \sqrt{21^2} = 21 \quad ; \quad \sqrt{(-33)^2} = 33}
\]
\subsection*{\color{myred} Remarque}
\noindent
La racine carrée d'un nombre négatif n'existe pas : la calculatrice affiche un message d'erreur.

\vspace{0.3cm}
\begin{center}
\fbox{%
  \parbox{6cm}{%
    \vspace{0.2cm}
    $\mathbf{\sqrt{-9}}$
    \rightline{\textcolor{red!70!black}{\textbf{\textsf{Math ERROR}}}}
    \vspace{0.1cm}
  }%
}
\end{center}
\subsection*{\color{myred} Application 1}
\noindent
Calculer la racine carrée des nombres suivants :
\[
\mathbf{\sqrt{4} \ ; \ \sqrt{9} \ ; \ \sqrt{16} \ ; \ \sqrt{25} \ ; \ \sqrt{36} \ ; \ \sqrt{49} \ ; \ \sqrt{64} \ ; \ \sqrt{81} \ ; \ \sqrt{100} \ ; \ \sqrt{121} \ ; \ \sqrt{144}}
\]
\subsection*{\color{myred} Correction }
\[
\mathbf{\sqrt{4} = 2 \quad ; \quad \sqrt{9} = 3 \quad ; \quad \sqrt{16} = 4 \quad ; \quad \sqrt{25} = 5 \quad ; \quad \sqrt{36} = 6 \quad ; \quad \sqrt{49} = 7}
\]
\[
\mathbf{\sqrt{64} = 8 \quad ; \quad \sqrt{81} = 9 \quad ; \quad \sqrt{100} = 10 \quad ; \quad \sqrt{121} = 11 \quad ; \quad \sqrt{144} = 12}
\]
\subsection*{\color{myred} Application 2 }
Calculer les expressions suivantes :
\begin{align*}
\boldsymbol{A = \left(\sqrt{2}\right)^2} \qquad \qquad \boldsymbol{B = \left(3\sqrt{3}\right)^2} \qquad \qquad \boldsymbol{C = \sqrt{(-11)^2} + \left(\sqrt{11}\right)^2} \\[0.3cm]
\boldsymbol{D = \left(1 + \sqrt{2}\right)^2} \hspace{7cm}
\end{align*}
\subsection*{\color{myred} Solution }

\vspace{0.3cm}
\begin{multicols}{2}
\noindent
$\boldsymbol{A = \left(\sqrt{2}\right)^2 = 2}$

\vspace{0.4cm}
\noindent
$\boldsymbol{B = \left(3\sqrt{3}\right)^2 = 3^2 \times \left(\sqrt{3}\right)^2}$
\\[0.1cm]
$\mathbf{= 9 \times 3 = 27}$

\columnbreak

\noindent
$\boldsymbol{C = \sqrt{(-11)^2} + \left(\sqrt{11}\right)^2}$
\\[0.1cm]
$\mathbf{= 11 + 11 = 22}$

\vspace{0.4cm}
\noindent
$\boldsymbol{D = \left(1 + \sqrt{2}\right)^2}$
\\[0.1cm]
$\mathbf{= 1^2 + 2 \times 1 \times \sqrt{2} + \left(\sqrt{2}\right)^2}$
\\[0.1cm]
$\mathbf{= 1 + 2\sqrt{2} + 2 = 3 + 2\sqrt{2}}$
\end{multicols}

\section*{\color{myred}II. Racines Carrées Et Opérations}

\subsection*{\color{myred} Activité 2 }
\begin{enumerate}[label=\textbf{\arabic*.}]
    \item Calculer : $\mathbf{\sqrt{9} \times \sqrt{16}}$ \quad et \quad $\mathbf{\sqrt{9 \times 16}}$. Que remarquez-vous ?
    
    \vspace{0.2cm}
    \item Calculer : $\mathbf{\sqrt{9} + \sqrt{16}}$ \quad et \quad $\mathbf{\sqrt{9 + 16}}$. Que remarquez-vous ?
    
    \vspace{0.2cm}
    \item Calculer : $\mathbf{\frac{\sqrt{64}}{\sqrt{4}}}$ \quad et \quad $\mathbf{\sqrt{\frac{64}{4}}}$. Que remarquez-vous ?
\end{enumerate}
\subsection*{\color{myred} 1. Produit de deux racines carrées }

\subsubsection*{\color{myred} Propriété }
Soient $a$ et $b$ deux nombres réels positifs. On a :
\[
\mathbf{\sqrt{a \times b} = \sqrt{a} \times \sqrt{b} \quad ; \quad \sqrt{a^2 \times b} = \sqrt{a^2} \times \sqrt{b} = a\sqrt{b}}
\]
\subsubsection*{\color{myred} Exemple }
\begin{align*}
\mathbf{A} &\mathbf{{}= \sqrt{12} = \sqrt{4 \times 3} = \sqrt{4} \times \sqrt{3} = 2 \times \sqrt{3} = 2\sqrt{3}} \\[0.2cm]
\mathbf{B} &\mathbf{{}= \sqrt{45} = \sqrt{9 \times 5} = \sqrt{9} \times \sqrt{5} = 3 \times \sqrt{5} = 3\sqrt{5}} \\[0.2cm]
\mathbf{C} &\mathbf{{}= \sqrt{72} = \sqrt{36 \times 2} = \sqrt{36} \times \sqrt{2} = 6 \times \sqrt{2} = 6\sqrt{2}} \\[0.2cm]
\mathbf{D} &\mathbf{{}= \sqrt{45} - 7\sqrt{5} + \sqrt{20} = \sqrt{9 \times 5} - 7\sqrt{5} + \sqrt{4 \times 5}} \\[0.1cm]
&\mathbf{{}= 3\sqrt{5} - 7\sqrt{5} + 2\sqrt{5} = -2\sqrt{5}}
\end{align*}
\subsection*{\color{myred} 2. Quotient de deux racines carrées }

\subsubsection*{\color{myred} Propriété}
Soient $a$ et $b$ deux nombres réels positifs, avec $b \neq 0$. On a :
\[
\mathbf{\sqrt{\frac{a}{b}} = \frac{\sqrt{a}}{\sqrt{b}} \quad ; \quad \sqrt{\frac{1}{b}} = \frac{1}{\sqrt{b}}}
\]
\subsubsection*{\color{myred} Exemple }
\[
\mathbf{A = \sqrt{\frac{25}{4}} = \frac{\sqrt{25}}{\sqrt{4}} = \frac{5}{2} \quad ; \quad B = \sqrt{\frac{1}{36}} = \frac{1}{\sqrt{36}} = \frac{1}{6}}
\]
\subsection*{\color{myred} Application 1}
Simplifier les expressions suivantes :

\vspace{0.3cm}
\begin{multicols}{2}
\noindent
$\mathbf{A = \sqrt{81} + 7\sqrt{3} - \sqrt{27}}$ \\[0.3cm]
$\mathbf{C = \sqrt{500} - 2\sqrt{5} + 3\sqrt{20}}$ \\[0.3cm]
$\mathbf{E = 5 + 6\sqrt{2}\left(3\sqrt{2} + 4\right)}$

\columnbreak

\noindent
$\mathbf{B = 2\sqrt{75} + \sqrt{27}}$ \\[0.3cm]
$\mathbf{D = \sqrt{125} - \sqrt{20} - \sqrt{45}}$
\end{multicols}
\subsection*{\color{myred} Correction }

\vspace{0.4cm}
\begin{multicols}{2}
\noindent
$\mathbf{A = \sqrt{81} + 7\sqrt{3} - \sqrt{9 \times 3}}$ \\[0.1cm]
$\mathbf{= 9 + 7\sqrt{3} - 3\sqrt{3}}$ \\[0.1cm]
$\mathbf{= 9 + 4\sqrt{3}}$

\vspace{0.4cm}
\noindent
$\mathbf{B = 2\sqrt{25 \times 3} + \sqrt{9 \times 3}}$ \\[0.1cm]
$\mathbf{= 2 \times 5\sqrt{3} + 3\sqrt{3}}$ \\[0.1cm]
$\mathbf{= 10\sqrt{3} + 3\sqrt{3} = 13\sqrt{3}}$

\vspace{0.4cm}
\noindent
$\mathbf{C = \sqrt{100 \times 5} - 2\sqrt{5} + 3\sqrt{4 \times 5}}$ \\[0.1cm]
$\mathbf{= 10\sqrt{5} - 2\sqrt{5} + 6\sqrt{5} = 14\sqrt{5}}$

\columnbreak

\noindent
$\mathbf{D = \sqrt{25 \times 5} - \sqrt{4 \times 5} - \sqrt{9 \times 5}}$ \\[0.1cm]
$\mathbf{= 5\sqrt{5} - 2\sqrt{5} - 3\sqrt{5}}$ \\[0.1cm]
$\mathbf{= 0}$

\vspace{0.4cm}
\noindent
$\mathbf{E = 5 + 6\sqrt{2} \times 3\sqrt{2} + 6\sqrt{2} \times 4}$ \\[0.1cm]
$\mathbf{= 5 + 18 \times \left(\sqrt{2}\right)^2 + 24\sqrt{2}}$ \\[0.1cm]
$\mathbf{= 5 + 36 + 24\sqrt{2}}$ \\[0.1cm]
$\mathbf{= 41 + 24\sqrt{2}}$
\end{multicols}
