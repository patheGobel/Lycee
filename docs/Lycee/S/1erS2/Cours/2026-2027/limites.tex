\subsubsection*{\underline{\textcolor{darkred}{ Interprétation :}}}

\begin{itemize}
\item On constate que lorsque $x$ prend des valeurs positives de plus en plus grandes, $f(x)$ prend des valeurs positives de plus en plus grandes. \\
On dit alors que la limite lorsque $x$ tend vers $+\infty$ de $f(x)$ est égale à $+\infty$. \\
On note :
\[
\boxed{\lim_{x \to +\infty} f(x) = +\infty}
\]

\item On constate que lorsque $x$ prend des valeurs positives de plus en plus grandes, $g(x)$ prend des valeurs positives de plus en plus proche de $0$. \\
On dit alors que la limite lorsque $x$ tend vers $+\infty$ de $g(x)$ est égale à $0$. \\
On note :
\[
\boxed{\lim_{x \to +\infty} g(x) = 0}
\]
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Théorème de l’unicité de la limite }}}
Soit $f$ une fonction.\\
Si $f$ admet une limite en un point $x_0$, en $+\infty$ ou en $-\infty$, alors cette limite est unique.

\subsection*{\underline{\textcolor{darkred}{ 2. Limites à l’infini des fonctions usuelles}}}
\subsubsection*{\underline{\textcolor{darkred}{ Propriété }}}

\noindent \textit{Soient $a \in \mathbb{R}$ et $n \in \mathbb{N} \setminus \{0\}$, on a :}

\vspace{0.4cm}

\begin{multicols}{2}
\noindent
1) $\displaystyle \lim_{x \to +\infty} a = a$ \\[0.5cm]
2) $\displaystyle \lim_{x \to +\infty} x^n = +\infty$ \\[0.5cm]
3) $\displaystyle \lim_{x \to +\infty} \frac{1}{x^n} = 0$

\columnbreak

\noindent
4) $\displaystyle \lim_{x \to -\infty} a = a$ \\[0.3cm]
5) $\displaystyle \lim_{x \to -\infty} x^n = \begin{cases} +\infty & \text{\textit{si $n$ est pair}} \\ -\infty & \text{\textit{si $n$ est impair}} \end{cases}$ \\[0.3cm]
6) $\displaystyle \lim_{x \to -\infty} \frac{1}{x^n} = 0$
\end{multicols}

\subsection*{\underline{\textcolor{darkred}{ 5. Limite à gauche et à droite en un point $x_{0}$ }}}
\subsubsection*{\underline{\textcolor{darkred}{ a. Définition }}}

\textit{Soit $f$ une fonction et $x_0 \in \mathbb{R}$.}

\begin{itemize}
\item \textit{On appelle limite à gauche en $x_0$ de $f$, la limite de $f$ lorsque $x$ tend vers $x_0$, en restant à gauche de $x_0$. On la note :}
\[
\lim_{x \to x_0^-} f(x) \quad \text{\textit{ou}} \quad \lim_{\substack{x \to x_0 \\ x < x_0}} f(x)
\]

\vspace{0.3cm}
\item \textit{On appelle limite à droite en $x_0$ de $f$, la limite de $f$ lorsque $x$ tend vers $x_0$, en restant à droite de $x_0$. On la note :}
\[
\lim_{x \to x_0^+} f(x) \quad \text{\textit{ou}} \quad \lim_{\substack{x \to x_0 \\ x > x_0}} f(x)
\]

\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}

Soit $f$ la fonction définie par : 
$
f(x) =
\begin{cases} 
2x - 1 & \text{\textit{si }} x > 1 \\ 
x^2 + x + 1 & \text{\textit{si }} x \leq 1 
\end{cases}
$

\vspace{0.3cm}

\noindent \textit{Calculons les limites en $1^-$ et en $1^+$ de $f$.} \\
\textit{On a :}

\begin{itemize}
    \item .
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 1^-} f(x) &= \lim_{x \to 1^-} x^2 + x + 1 \\
    &= (1)^2 + (1) + 1 \\
    &= 3
    \end{aligned}
    $
    
    $\boxed{\lim_{x \to 1^-} f(x) = 3}$
    
    \vspace{0.3cm}
    \item .
    
     $
    \begin{aligned}
    \displaystyle \lim_{x \to 1^+} f(x) &= \lim_{x \to 1^+} 2x - 1 \\
    &= 2(1) - 1 \\
    &= 1
    \end{aligned}
    $
    
    $ \boxed{\lim_{x \to 1^+} f(x) = 1}$
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ b. Propriété }}}

Pour tout réel $a$, on a :
\[
\lim_{x \to a^+} \frac{1}{x - a} = +\infty \quad ; \quad \lim_{x \to a^-} \frac{1}{x - a} = -\infty
\]

\begin{center}
\begin{tikzpicture}[node style/.style={fill opacity=0,text opacity=1}]
        \tkzTabInit[espcl=2,lgt=3]{$x$/.5,$x - a$/.7}{$-\infty$,$a$ ,$+\infty$}
        \tkzTabLine{ ,-,z,+, }
\end{tikzpicture}
\end{center}

\paragraph{\underline{Explication :}}
Le tableau de signes de l'expression $x - a$ permet de déterminer si le dénominateur tend vers $0$ par valeurs positives ($0^+$) ou par valeurs négatives ($0^-$) :

\begin{itemize}
    \item \textbf{Limite à droite ($x \to a^+$) :} 
    Lorsque $x > a$ (à droite de $a$), le tableau indique que $x - a$ est \textbf{positif} ($+$). Le dénominateur tend donc vers $0$ par valeurs supérieures, soit $0^+$.
    \[
    \text{Par règle des signes : } \lim_{x \to a^+} \frac{1}{x - a} = \frac{1}{0^+} = +\infty
    \]

    \item \textbf{Limite à gauche ($x \to a^-$) :} 
    Lorsque $x < a$ (à gauche de $a$), le tableau indique que $x - a$ est \textbf{négatif} ($-$). Le dénominateur tend donc vers $0$ par valeurs inférieures, soit $0^-$.
    \[
    \text{Par règle des signes : } \lim_{x \to a^-} \frac{1}{x - a} = \frac{1}{0^-} = -\infty
    \]
\end{itemize}
\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}    

\noindent
\begin{multicols}{2}

\noindent \textbf{1. Limites de $\dfrac{1}{x}$ en $0$ :}

\begin{center}
\begin{tikzpicture}[node style/.style={fill opacity=0,text opacity=1}]
    \tkzTabInit[espcl=1.5,lgt=2]{$x$/.5,$x$/.7}{$-\infty$,$0$,$+\infty$}
    \tkzTabLine{ ,-,z,+, }
\end{tikzpicture}
\end{center}

\begin{itemize}
    \item \textbf{À droite ($x \to 0^+$) :} d'après le tableau, pour $x > 0$, le dénominateur est positif ($+$). Donc $x \to 0^+$.
    \[
    \lim_{x \to 0^+} \frac{1}{x} = \frac{1}{0^+} = +\infty
    \]
    \item \textbf{À gauche ($x \to 0^-$) :} d'après le tableau, pour $x < 0$, le dénominateur est négatif ($-$). Donc $x \to 0^-$.
    \[
    \lim_{x \to 0^-} \frac{1}{x} = \frac{1}{0^-} = -\infty
    \]
\end{itemize}

\columnbreak

\noindent \textbf{2. Limites de $\dfrac{1}{x - 2}$ en $2$ :}

    \]

\subsection*{\underline{\textcolor{darkred}{ 6. Existence de limite d’une fonction en un point $x_{0}$ }}}
\subsubsection*{\underline{\textcolor{darkred}{ b. Propriété }}}
Soit $f$ une fonction et $x_0 \in \mathbb{R}$. La fonction $f$ admet une limite $l$ en $x_0$ si et seulement si on a :
\[
\lim_{x \to x_0^+} f(x) = \lim_{x \to x_0^-} f(x) = l
\]

\vspace{0.3cm}
\noindent{\Large\bfseries\color{darkred} Méthode :}

Pour déterminer si une fonction $f$ admet ou non une limite en un point $x_0$, on calcule la limite à gauche en $x_0$ et la limite à droite en $x_0$ de $f$, puis on compare les résultats.
\begin{itemize}
\item Si $\displaystyle \lim_{x \to x_0^+} f(x) = \lim_{x \to x_0^-} f(x)$, alors $f$ admet une limite $l$ en $x_0$ vérifiant :
\[
l = \lim_{x \to x_0^+} f(x) = \lim_{x \to x_0^-} f(x).
\]

\vspace{0.2cm}
\item Si $\displaystyle \lim_{x \to x_0^+} f(x) \neq \lim_{x \to x_0^-} f(x)$, alors $f$ n'admet pas de limite en $x_0$.
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}  

\noindent 1. \textit{Soit $f$ la fonction définie par : } 
$\begin{cases} f(x) = x - 1 & \text{\textit{si }} x > 0 \\ f(x) = x^2 + 1 & \text{\textit{si }} x \leq 0 \end{cases}.$

\vspace{0.1cm}

\noindent \textit{La fonction $f$ admet-elle une limite en $0$ ?} \\
\textit{On a :}

\begin{itemize}
    \item $\displaystyle \lim_{x \to 0^-} f(x) = \lim_{x \to 0^-} x^2 + 1 = (0)^2 + 1 = 1$
    \vspace{0.2cm}
    \item $\displaystyle \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} x - 1 = (0) - 1 = -1$
\end{itemize}

\noindent \textit{Ainsi, $\displaystyle \lim_{x \to 0^+} f(x) \neq \lim_{x \to 0^-} f(x)$, donc $f$ n'admet pas de limite en $0$.}

\vspace{0.6cm}

\noindent 2. \textit{Soit $g$ la fonction définie par : } 
$\begin{cases} g(x) = x - 1 & \text{\textit{si }} x > 1 \\ g(x) = x^2 + x - 2 & \text{\textit{si }} x \leq 1 \end{cases}.$

\vspace{0.1cm}

\noindent \textit{La fonction $g$ admet-elle une limite en $1$ ?} \\
\textit{On a :}

\begin{itemize}
    \item $\displaystyle \lim_{x \to 1^-} g(x) = \lim_{x \to 1^-} x^2 + x - 2 = (1)^2 + (1) - 2 = 0$
    \vspace{0.2cm}
    \item $\displaystyle \lim_{x \to 1^+} g(x) = \lim_{x \to 1^+} x - 1 = (1) - 1 = 0$
\end{itemize}

\noindent \textit{Ainsi, $\displaystyle \lim_{x \to 1^+} g(x) = \lim_{x \to 1^-} g(x) = 0$, d'où $g$ admet une limite en $1$ et on a :}
\[
\lim_{x \to 1} g(x) = 0.
\]

\begin{itemize}

    \vspace{0.2cm}
    
    \textit{Étudions le signe de $x - 1$.} \\
    $x - 1 = 0 \implies x = 1$
    
    \vspace{0.2cm}
    
    \begin{center}
    \begin{tabular}{|c|ccccc|}
        \hline
        $x$ & $-\infty$ & & $1$ & & $+\infty$ \\
        \hline
        $x - 1$ & & $-$ & $0$ & $+$ & \\
        \hline
    \end{tabular}
    \end{center}
    
    \vspace{0.2cm}
    
    \[
    \lim_{x \to 1^+} \frac{x - 3}{x - 1} :
    \begin{cases}
    \displaystyle \lim_{x \to 1^+} (x - 3) = -2 \\
    & \text{Par quotient} \\
    \displaystyle \lim_{x \to 1^+} (x - 1) = 0^+
    \end{cases}
    \]
    
    \[
    \boxed{\lim_{x \to 1^+} \frac{x - 3}{x - 1} = -\infty}
    \]

    \vspace{0.4cm}

    \item \textit{Calculons $\displaystyle \lim_{x \to +\infty} \frac{x + 1}{x - 1}$ :}
    
    \vspace{0.2cm}
    
    \[
    \lim_{x \to +\infty} \frac{x + 1}{x - 1} :
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} (x + 1) = +\infty \\
    & \text{Par quotient (F.I)} \\
    \displaystyle \lim_{x \to +\infty} (x - 1) = +\infty
    \end{cases}
    \]
    
    \vspace{0.2cm}
    \textit{C'est une Forme Indéterminée (F.I) ; on ne peut pas conclure directement.}

    \vspace{0.4cm}

    \item \textit{Calculons $\displaystyle \lim_{x \to 1} \frac{x^2 + 1}{x - 3}$ :}
    
    \vspace{0.2cm}
    
    \[
    \lim_{x \to 1} \frac{x^2 + 1}{x - 3} :
    \begin{cases}
    \displaystyle \lim_{x \to 1} (x^2 + 1) = 2 \\
    & \text{Par quotient} \\
    \displaystyle \lim_{x \to 1} (x - 3) = -2
    \end{cases}
    \]
    
    \[
    \boxed{\lim_{x \to 1} \frac{x^2 + 1}{x - 3} = -1}
    \]
\end{enumerate}

\subsubsection*{\underline{\textcolor{darkred}{ 4. Limites de $\sqrt{u}$ et $|u|$} }}
\subsubsection*{\underline{\textcolor{darkred}{ a. Propriété }}}

Soit $u$ une fonction positive et $x_0$ un réel.

\vspace{0.3cm}

\begin{itemize}
\item Si $\displaystyle \lim_{x \to x_0} u = l$, alors $\displaystyle \lim_{x \to x_0} \sqrt{u} = \sqrt{l}$
\item Si $\displaystyle \lim_{x \to x_0} u = +\infty$, alors $\displaystyle \lim_{x \to x_0} \sqrt{u} = +\infty$
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}} 

\begin{enumerate}
    \item \textit{Soit $g$ la fonction définie par : $g(x) = \sqrt{x + 3}$. Calculons $\displaystyle \lim_{x \to 1} g(x)$.}
    
    \vspace{0.2cm}
    
    \textit{On a : $\displaystyle \lim_{x \to 1} x + 3 = 4$ donc $\displaystyle \lim_{x \to 1} g(x) = \sqrt{4} = 2$.}
    
    \[
    \boxed{\lim_{x \to 1} g(x) = 2}
    \]
\end{enumerate}

\begin{enumerate}
    \setcounter{enumi}{1}
    \item \textit{Soit $f$ la fonction définie par : $f(x) = \sqrt{x + 1}$. Calculons $\displaystyle \lim_{x \to +\infty} f(x)$.}
    
    \vspace{0.2cm}
    
    \textit{On a : $\displaystyle \lim_{x \to +\infty} x + 1 = +\infty$ donc $\displaystyle \lim_{x \to +\infty} f(x) = +\infty$.}
    
    \[
    \boxed{\lim_{x \to +\infty} f(x) = +\infty}
    \]
\end{enumerate}

\subsubsection*{\underline{\textcolor{darkred}{ Exercice }}}
Calculer les limites suivantes :
\begin{enumerate}
    \item \textit{Calculons $\displaystyle \lim_{x \to -\infty} \sqrt{x^2 + 1}$ :}
    
    \vspace{0.2cm}
    
    \textit{On a : $\displaystyle \lim_{x \to -\infty} (x^2 + 1) = +\infty$ donc $\displaystyle \lim_{x \to -\infty} \sqrt{x^2 + 1} = +\infty$.}
    
    \[
    \boxed{\lim_{x \to -\infty} \sqrt{x^2 + 1} = +\infty}
    \]

    \vspace{0.4cm}

    \item \textit{Calculons $\displaystyle \lim_{x \to 1} \sqrt{x - 1}$ :}
    
    \vspace{0.2cm}
    
    \textit{On a : $\displaystyle \lim_{x \to 1} (x - 1) = 0$ donc $\displaystyle \lim_{x \to 1} \sqrt{x - 1} = 0$.}
    
    \[
    \boxed{\lim_{x \to 1} \sqrt{x - 1} = 0}
    \]

    \vspace{0.4cm}

    \item \textit{Calculons $\displaystyle \lim_{x \to 2} |x - 3|$ :}
    
    \vspace{0.2cm}

    \]
\end{enumerate}

\section*{\underline{\textcolor{darkred}{ III. Calculs de limites de certaines fonctions}}}
\subsection*{\underline{\textcolor{darkred}{ 1. Limites à l’infini des fonctions polynômes }}}
\subsubsection*{\underline{\textcolor{darkred}{ Propriété }}}

La limite à l'infini d'une fonction polynôme est égale à la limite de son monôme du plus haut degré.

\vspace{0.3cm}

\noindent En effet, on a $f(x) = a_n x^n + a_{n-1} x^{n-1} + \dots + a_0$, avec $a_n$, $a_{n-1}$, \dots, $a_0$ des réels tels que $a_n \neq 0$.

\begin{align*}
\lim_{x \to \pm\infty} f(x) &= \lim_{x \to \pm\infty} (a_n x^n + a_{n-1} x^{n-1} + \dots + a_0) \\
&= \lim_{x \to \pm\infty} a_n x^n \left(1 + \frac{a_{n-1}}{a_n x} + \dots + \frac{a_0}{a_n x^n}\right) \\
&= \lim_{x \to \pm\infty} a_n x^n \times \lim_{x \to \pm\infty} \left(1 + \frac{a_{n-1}}{a_n x} + \dots + \frac{a_0}{a_n x^n}\right) \\
&= \lim_{x \to \pm\infty} a_n x^n \quad \text{car} \quad \lim_{x \to \pm\infty} \left(1 + \frac{a_{n-1}}{a_n x} + \dots + \frac{a_0}{a_n x^n}\right) \\
&= 1
\end{align*}

\noindent Donc $\displaystyle \lim_{x \to \pm\infty} f(x) = \lim_{x \to \pm\infty} a_n x^n$.


 Méthode :

\vspace{0.1cm}

Pour calculer la limite à l'infini d'une fonction polynôme, il suffit de calculer la limite de son monôme du plus haut degré.

\vspace{0.4cm}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}} 

Calculons les limites suivantes

\vspace{0.2cm}

\begin{enumerate}
    \item 
    
    $\begin{aligned}[t]
        \lim_{x \to -\infty} (-2x^3 + 4x^2 + 6x - 1) &= \lim_{x \to -\infty} -2x^3 \\
        &= -2(-\infty)\\ 
        &= +\infty
    \end{aligned}$
    
    \boxed{\lim_{x \to -\infty} (-2x^3 + 4x^2 + 6x - 1) = +\infty}
    
    \vspace{0.4cm}
    
    \item 
    
    $\begin{aligned}[t]
        \lim_{x \to +\infty} (4x^4 + x^3 + x - 1) &= \lim_{x \to +\infty} 4x^4 \\
        &= 4(+\infty)\\ 
        &= +\infty
    \end{aligned}$
    
    \boxed{\lim_{x \to +\infty} (4x^4 + x^3 + x - 1) = +\infty}
\end{enumerate}

\subsection*{\underline{\textcolor{darkred}{ 2. Limites à l’infini des fonctions rationnelles }}}
\subsubsection*{\underline{\textcolor{darkred}{ Propriété }}}

La limite à l'infini d'une fonction rationnelle est égale à la limite du quotient des monômes du plus haut degré du numérateur et du dénominateur.


\vspace{0.3cm}

\noindent En effet :

\noindent Soit $f$ une fonction rationnelle définie par : $f(x) = \dfrac{a_n x^n + a_{n-1} x^{n-1} + \dots + a_0}{b_m x^m + b_{m-1} x^{m-1} + \dots + b_0}$ avec $a_n \neq 0$ et $b_m \neq 0$. On a :

\begin{align*}
\lim_{x \to \pm\infty} f(x) &= \lim_{x \to \pm\infty} \frac{a_n x^n + a_{n-1} x^{n-1} + \dots + a_0}{b_m x^m + b_{m-1} x^{m-1} + \dots + b_0} \\
&= \lim_{x \to \pm\infty} \frac{a_n x^n \left(1 + \dfrac{a_{n-1}}{a_n x} + \dots + \dfrac{a_0}{a_n x^n}\right)}{b_m x^m \left(1 + \dfrac{b_{m-1}}{b_m x} + \dots + \dfrac{b_0}{b_m x^m}\right)} \\
&= \lim_{x \to \pm\infty} \frac{a_n x^n}{b_m x^m} \left( \frac{1 + \dfrac{a_{n-1}}{a_n x} + \dots + \dfrac{a_0}{a_n x^n}}{1 + \dfrac{b_{m-1}}{b_m x} + \dots + \dfrac{b_0}{b_m x^m}} \right) \\
&= \lim_{x \to \pm\infty} \frac{a_n x^n}{b_m x^m} \times \lim_{x \to \pm\infty} \left( \frac{1 + \dfrac{a_{n-1}}{a_n x} + \dots + \dfrac{a_0}{a_n x^n}}{1 + \dfrac{b_{m-1}}{b_m x} + \dots + \dfrac{b_0}{b_m x^m}} \right)
\end{align*}

\noindent Or $\displaystyle \lim_{x \to \pm\infty} \left( \frac{1 + \dfrac{a_{n-1}}{a_n x} + \dots + \dfrac{a_0}{a_n x^n}}{1 + \dfrac{b_{m-1}}{b_m x} + \dots + \dfrac{b_0}{b_m x^m}} \right) = 1$ donc $\displaystyle \lim_{x \to \pm\infty} f(x) = \lim_{x \to \pm\infty} \frac{a_n x^n}{b_m x^m}$.

\subsubsection*{\underline{\textcolor{darkred}{ Méthode : }}}
Pour calculer la limite à l'infini d'une fonction rationnelle, il suffit de calculer la limite du quotient des monômes du plus haut degré du numérateur et du dénominateur.

\vspace{0.4cm}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple : }}}
Calculons les limites suivantes

\vspace{0.3cm}

\begin{enumerate}
    \item 
    $\begin{aligned}[t]
        \lim_{x \to -\infty} \frac{-2x^3 + x - 1}{x^2 - x + 2} &= \lim_{x \to -\infty} \frac{-2x^3}{x^2} \\
        &= \lim_{x \to -\infty} -2x \\
        &= -2(-\infty) \\
        &= +\infty
    \end{aligned}$
    
    \vspace{0.2cm}
    \boxed{\lim_{x \to -\infty} \frac{-2x^3 + x - 1}{x^2 - x + 2} = +\infty}

    \vspace{0.6cm}

    \item 
    $\begin{aligned}[t]
        \lim_{x \to +\infty} \frac{x^3 + 4x^2 + 6x - 1}{2x^5 - 6x + 2} &= \lim_{x \to +\infty} \frac{x^3}{2x^5} \\
        &= \lim_{x \to +\infty} \frac{1}{2x^2} \\
        &= 0
    \end{aligned}$
    
    \vspace{0.2cm}
    \boxed{\lim_{x \to +\infty} \frac{x^3 + 4x^2 + 6x - 1}{2x^5 - 6x + 2} = 0}

    \vspace{0.6cm}

    \item 
    $\begin{aligned}[t]
        \lim_{x \to +\infty} \frac{4x^2 - 2x - 3}{2x^2 - x - 5} &= \lim_{x \to +\infty} \frac{4x^2}{2x^2} \\
        &= 2
    \end{aligned}$
    
    \vspace{0.2cm}
    \boxed{\lim_{x \to +\infty} \frac{4x^2 - 2x - 3}{2x^2 - x - 5} = 2}
\end{enumerate}

\subsection*{\underline{\textcolor{darkred}{3. Limite en $x_0$ de $\frac{u(x)}{v(x)}$ avec $u(x_0)=v(x_0)=0$}}}

\subsubsection*{\underline{\textcolor{darkred}{ Méthode : }}}

Pour calculer la limite en un point $x_0$ d'une fonction du type : $f = \dfrac{u}{v}$ où $u(x_0) = v(x_0) = 0$, on cherche, d'abord, à mettre en facteur $(x - x_0)$ au numérateur et dénominateur, puis on simplifie $f(x)$. \\
Ensuite, on recalcule la limite de $f$ avec son expression simplifiée.

\subsubsection*{\underline{\textcolor{darkred}{ Exemple : }}}

\vspace{0.3cm}

\begin{enumerate}
    \item \textit{Soit $f$ la fonction définie par $f(x) = \dfrac{x^2 + 6x - 7}{x^2 + 2x - 3}$. Calculons $\displaystyle \lim_{x \to 1} f(x)$.}
    
    \vspace{0.1cm}
    
    $
    \displaystyle \lim_{x \to 1} f(x):
    \begin{cases}
    \displaystyle \lim_{x \to 1} (x^2 + 6x - 7) = 0\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to 1} (x^2 + 2x - 3) = 0
    \end{cases}
    $
 
    \textit{Levons l'indétermination :}
    
    Factorisons $ x^2 + 6x - 7 $  et $x^2 + 2x - 3 $
    
    On a : $ x^2 + 6x - 7 = (x - 1)(x + 7)$  et $x^2 + 2x - 3 = (x - 1)(x + 3)$
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 1} f(x) & =\displaystyle \lim_{x \to 1} \dfrac{x^2 + 6x - 7}{x^2 + 2x - 3}\\
    									 &=\displaystyle \lim_{x \to 1} \dfrac{(x - 1)(x + 7)}{(x - 1)(x + 3)}\\
    									 &=\displaystyle \lim_{x \to 1} \dfrac{x + 7}{x + 3}\\
    									 &= \frac{1 + 7}{1 + 3}\\
    									 &= 2
    \end{aligned}
    $
    
\text{\textit{Donc, }} \boxed{\lim_{x \to 1} f(x) = 2}

    \vspace{0.6cm}

    \item \textit{Soit $g$ la fonction définie par $g(x) = \dfrac{\sqrt{x + 3} - 2}{x - 1}$. Calculons $\displaystyle \lim_{x \to 1} g(x)$.}
    
    \vspace{0.1cm}
    
    $
    \displaystyle \lim_{x \to 1} g(x):
    \begin{cases}
    \displaystyle \lim_{x \to 1} (\sqrt{x + 3} - 2) = 0\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to 1} (x - 1) = 0
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    Utilisons l'expression conjuguée du numérateur :
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 1} g(x) &= \displaystyle \lim_{x \to 1} \dfrac{\sqrt{x + 3} - 2}{x - 1}\\
    &= \displaystyle \lim_{x \to 1} \dfrac{(\sqrt{x + 3} - 2)(\sqrt{x + 3} + 2)}{(x - 1)(\sqrt{x + 3} + 2)}\\
    &= \displaystyle \lim_{x \to 1} \dfrac{(\sqrt{x + 3})^2 - 2^2}{(x - 1)(\sqrt{x + 3} + 2)}\\
    &= \displaystyle \lim_{x \to 1} \dfrac{x + 3 - 4}{(x - 1)(\sqrt{x + 3} + 2)}\\
    &= \displaystyle \lim_{x \to 1} \dfrac{x - 1}{(x - 1)(\sqrt{x + 3} + 2)}\\
    &= \displaystyle \lim_{x \to 1} \dfrac{1}{\sqrt{x + 3} + 2}\\
    &= \frac{1}{\sqrt{1 + 3} + 2}\\
    &= \frac{1}{4}
    \end{aligned}
    $
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to 1} g(x) = \frac{1}{4}}

    \vspace{0.6cm}

    \item \textit{Soit $h$ la fonction définie par $h(x) = \dfrac{x - 2}{\sqrt{x - 2}}$. Calculons $\displaystyle \lim_{x \to 2^+} h(x)$.}
    
    \vspace{0.1cm}
    
    $
    \displaystyle \lim_{x \to 2^+} h(x):
    \begin{cases}
    \displaystyle \lim_{x \to 2^+} (x - 2) = 0\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to 2^+} \sqrt{x - 2} = 0
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    Rendons rationnel le dénominateur :
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 2^+} h(x) &= \displaystyle \lim_{x \to 2^+} \dfrac{x - 2}{\sqrt{x - 2}}\\
    &= \displaystyle \lim_{x \to 2^+} \dfrac{(x - 2)\sqrt{x - 2}}{(\sqrt{x - 2})^2}\\
    &= \displaystyle \lim_{x \to 2^+} \dfrac{(x - 2)\sqrt{x - 2}}{x - 2}\\
    &= \displaystyle \lim_{x \to 2^+} \sqrt{x - 2}\\
    &= 0
    \end{aligned}
    $
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to 2^+} h(x) = 0}
\end{enumerate}

\subsubsection*{\underline{\textcolor{darkred}{ Exercice : }}}

Calculer les limites suivantes :
\vspace{0.3cm}

\begin{enumerate}
    \item 
    $
    \displaystyle \lim_{x \to 2} \frac{x^2 - x - 2}{x^2 - 5x + 6}:
    \begin{cases}
    \displaystyle \lim_{x \to 2} (x^2 - x - 2) = 0\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to 2} (x^2 - 5x + 6) = 0
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    Factorisons $x^2 - x - 2$ et $x^2 - 5x + 6$ :
    
    On a : $x^2 - x - 2 = (x - 2)(x + 1)$ et $x^2 - 5x + 6 = (x - 2)(x - 3)$.
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 2} \frac{x^2 - x - 2}{x^2 - 5x + 6} &= \displaystyle \lim_{x \to 2} \frac{(x - 2)(x + 1)}{(x - 2)(x - 3)}\\
    &= \displaystyle \lim_{x \to 2} \frac{x + 1}{x - 3}\\
    &= \frac{2 + 1}{2 - 3}\\
    &= -3
    \end{aligned}
    $
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to 2} \frac{x^2 - x - 2}{x^2 - 5x + 6} = -3}

    \vspace{0.6cm}

    \item 
    $
    \displaystyle \lim_{x \to 1} \frac{\sqrt{x + 3} - \sqrt{2x + 2}}{x - 1}:
    \begin{cases}
    \displaystyle \lim_{x \to 1} (\sqrt{x + 3} - \sqrt{2x + 2}) = 0\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to 1} (x - 1) = 0
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    Utilisons l'expression conjuguée du numérateur :
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 1} \frac{\sqrt{x + 3} - \sqrt{2x + 2}}{x - 1} &= \displaystyle \lim_{x \to 1} \frac{(\sqrt{x + 3} - \sqrt{2x + 2})(\sqrt{x + 3} + \sqrt{2x + 2})}{(x - 1)(\sqrt{x + 3} + \sqrt{2x + 2})}\\
    &= \displaystyle \lim_{x \to 1} \frac{(\sqrt{x + 3})^2 - (\sqrt{2x + 2})^2}{(x - 1)(\sqrt{x + 3} + \sqrt{2x + 2})}\\
    &= \displaystyle \lim_{x \to 1} \frac{(x + 3) - (2x + 2)}{(x - 1)(\sqrt{x + 3} + \sqrt{2x + 2})}\\
    &= \displaystyle \lim_{x \to 1} \frac{-x + 1}{(x - 1)(\sqrt{x + 3} + \sqrt{2x + 2})}\\
    &= \displaystyle \lim_{x \to 1} \frac{-(x - 1)}{(x - 1)(\sqrt{x + 3} + \sqrt{2x + 2})}\\
    &= \displaystyle \lim_{x \to 1} \frac{-1}{\sqrt{x + 3} + \sqrt{2x + 2}}\\
    &= \frac{-1}{\sqrt{1 + 3} + \sqrt{2 + 2}}\\
    &= -\frac{1}{4}
    \end{aligned}
    $
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to 1} \frac{\sqrt{x + 3} - \sqrt{2x + 2}}{x - 1} = -\frac{1}{4}}

    \vspace{0.6cm}

    \item 
    $
    \displaystyle \lim_{x \to 0^+} \frac{x - \sqrt{x}}{x}:
    \begin{cases}
    \displaystyle \lim_{x \to 0^+} (x - \sqrt{x}) = 0\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to 0^+} x = 0
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    Mettons $\sqrt{x}$ ou $x$ en facteur au numérateur (on sait que $x = (\sqrt{x})^2$ pour $x > 0$) :
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 0^+} \frac{x - \sqrt{x}}{x} &= \displaystyle \lim_{x \to 0^+} \left( \frac{x}{x} - \frac{\sqrt{x}}{x} \right)\\
    &= \displaystyle \lim_{x \to 0^+} \left( 1 - \frac{1}{\sqrt{x}} \right)
    \end{aligned}
    $
    
    Or $\displaystyle \lim_{x \to 0^+} \sqrt{x} = 0^+$, donc $\displaystyle \lim_{x \to 0^+} \frac{1}{\sqrt{x}} = +\infty$.
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to 0^+} \frac{x - \sqrt{x}}{x} = -\infty}

    \vspace{0.6cm}

    \item 
    $
    \displaystyle \lim_{x \to 4} \frac{\sqrt{x} - 2}{x - 4}:
    \begin{cases}
    \displaystyle \lim_{x \to 4} (\sqrt{x} - 2) = 0\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to 4} (x - 4) = 0
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    Utilisons l'expression conjuguée du numérateur :
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 4} \frac{\sqrt{x} - 2}{x - 4} &= \displaystyle \lim_{x \to 4} \frac{(\sqrt{x} - 2)(\sqrt{x} + 2)}{(x - 4)(\sqrt{x} + 2)}\\
    &= \displaystyle \lim_{x \to 4} \frac{x - 4}{(x - 4)(\sqrt{x} + 2)}\\
    &= \displaystyle \lim_{x \to 4} \frac{1}{\sqrt{x} + 2}\\
    &= \frac{1}{\sqrt{4} + 2}\\
    &= \frac{1}{4}
    \end{aligned}
    $
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to 4} \frac{\sqrt{x} - 2}{x - 4} = \frac{1}{4}}
\end{enumerate}

\subsection*{\underline{\textcolor{darkred}{4. Limites à l'infini des fonctions du type : $f = \sqrt{p} + q$, où $p$ et $q$ des polynômes}}}
\vspace{0.2cm}

\subsubsection*{\underline{\textcolor{darkred}{ Méthode : }}}

Soit $f$ une fonction du type : $f = \sqrt{p} + q$, avec $p$ et $q$ des fonctions polynômes
\begin{itemize}
    \item Si le calcul direct de la limite de $f$ donne une forme indéterminée, alors on peut lever l'indétermination en réécrivant $f$ sous la forme : $f(x) = |x|\sqrt{\dfrac{p(x)}{x^2}} + q(x)$, ensuite mettre $x$ en facteur puis calculer la limite du produit obtenu.
    \item Si on voit qu'en mettant $x$ en facteur, le calcul de la limite du produit donnerait une forme indéterminée, alors on peut réécrire $f$ sous la forme :\\
    $f(x) = \dfrac{(\sqrt{p} + q)(\sqrt{p} - q)}{\sqrt{p} - q}$, ensuite mettre $x$ en facteur au numérateur et au dénominateur, puis simplifier $x$, si nécessaire, avant de calculer la limite du quotient obtenu.
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple : }}}

\begin{enumerate}
    \item \textit{Soit $f$ la fonction définie par $f(x) = \sqrt{4x^2 - 3x + 1} + x + 2$. Calculons $\displaystyle \lim_{x \to -\infty} f(x)$.}
    
    \vspace{0.1cm}
    
    $
    \displaystyle \lim_{x \to -\infty} f(x):
    \begin{cases}
    \displaystyle \lim_{x \to -\infty} \sqrt{4x^2 - 3x + 1} = +\infty\\[3mm]
    &\textit{donc la somme est une forme indéterminée.}\\
    \displaystyle \lim_{x \to -\infty} (x + 2) = -\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to -\infty} \sqrt{4x^2 - 3x + 1} + x + 2 &= \lim_{x \to -\infty} |x|\sqrt{4 - \frac{3}{x} + \dfrac{1}{x^2}} + x + 2 \\
        &=\lim_{x \to -\infty} -x\sqrt{4 - \frac{3}{x} + \frac{1}{x^2}} + x + 2 \quad (|x| = -x \text{ car } x < 0) \\
        &=\lim_{x \to -\infty} x \left( -\sqrt{4 - \dfrac{3}{x} + \dfrac{1}{x^2}} + 1 + \dfrac{2}{x} \right)\\
        &=\begin{cases}
    \displaystyle \lim_{x \to -\infty} x = -\infty \\[3mm]
    &\text{ Par produit }\\
    \displaystyle \lim_{x \to -\infty} \left( -\sqrt{4 - \frac{3}{x} + \frac{1}{x^2}} + 1 + \frac{2}{x} \right) = -1
    \end{cases}
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to -\infty} f(x) = +\infty}
    
    \item \textit{Soit $g$ la fonction définie par $g(x) = \sqrt{x^2 + 1} - x - 1$. Calculons $\displaystyle \lim_{x \to +\infty} g(x)$.}
    
    \vspace{0.1cm}
    
    $
    \displaystyle \lim_{x \to +\infty} g(x):
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} \sqrt{x^2 + 1} = +\infty\\[3mm]
    &\textit{donc la somme est une forme indéterminée.}\\
    \displaystyle \lim_{x \to +\infty} (-x - 1) = -\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to +\infty} \sqrt{x^2 + 1} - x - 1 &= \lim_{x \to +\infty} \frac{(\sqrt{x^2 + 1} - (x + 1))(\sqrt{x^2 + 1} + (x + 1))}{\sqrt{x^2 + 1} + x + 1} \\
        &= \lim_{x \to +\infty} \frac{(\sqrt{x^2 + 1})^2 - (x + 1)^2}{\sqrt{x^2 + 1} + x + 1} \\
        &= \lim_{x \to +\infty} \frac{-2x}{\sqrt{x^2 + 1} + x + 1} \\
        &= \lim_{x \to +\infty} \frac{-2x}{x \left( \sqrt{1 + \dfrac{1}{x^2}} + 1 + \dfrac{1}{x} \right)} \quad (|x| = x \text{ car } x > 0) \\
        &= \lim_{x \to +\infty} \frac{-2}{\sqrt{1 + \dfrac{1}{x^2}} + 1 + \dfrac{1}{x}} \\
        &= \begin{cases}
    \displaystyle \lim_{x \to +\infty} -2 = -2 \\[3mm]
    &\text{ Par quotient }\\
    \displaystyle \lim_{x \to +\infty} \left( \sqrt{1 + \frac{1}{x^2}} + 1 + \frac{1}{x} \right) = 2
    \end{cases}
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to +\infty} g(x) = -1}
\end{enumerate}

\subsubsection*{\underline{\textcolor{darkred}{ Exercice : }}}
Calculer les limites suivantes :
\begin{enumerate}
    \item 
    $
    \displaystyle \lim_{x \to -\infty} x + \sqrt{4x^2 + 1}:
    \begin{cases}
    \displaystyle \lim_{x \to -\infty} \sqrt{4x^2 + 1} = +\infty\\[3mm]
    &\textit{donc la somme est une forme indéterminée.}\\
    \displaystyle \lim_{x \to -\infty} x = -\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to -\infty} x + \sqrt{4x^2 + 1} &= \lim_{x \to -\infty} x + |x|\sqrt{4 + \dfrac{1}{x^2}} \\
        &= \lim_{x \to -\infty} x - x\sqrt{4 + \dfrac{1}{x^2}} \quad (|x| = -x \text{ car } x < 0) \\
        &= \lim_{x \to -\infty} x \left( 1 - \sqrt{4 + \dfrac{1}{x^2}} \right) \\
        &= \begin{cases}
    \displaystyle \lim_{x \to -\infty} x = -\infty \\[3mm]
    &\text{ Par produit }\\
    \displaystyle \lim_{x \to -\infty} \left( 1 - \sqrt{4 + \frac{1}{x^2}} \right) = 1 - 2 = -1
    \end{cases}
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to -\infty} (x + \sqrt{4x^2 + 1}) = +\infty}

    \vspace{0.6cm}

    \item 
    $
    \displaystyle \lim_{x \to +\infty} x + 1 - \sqrt{x^2 + 2x + 1}:
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} (x + 1) = +\infty\\[3mm]
    &\textit{donc la somme est une forme indéterminée.}\\
    \displaystyle \lim_{x \to +\infty} -\sqrt{x^2 + 2x + 1} = -\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to +\infty} x + 1 - \sqrt{x^2 + 2x + 1} &= \lim_{x \to +\infty} \frac{((x + 1) - \sqrt{x^2 + 2x + 1})((x + 1) + \sqrt{x^2 + 2x + 1})}{x + 1 + \sqrt{x^2 + 2x + 1}} \\
        &= \lim_{x \to +\infty} \frac{(x + 1)^2 - (x^2 + 2x + 1)}{x + 1 + \sqrt{x^2 + 2x + 1}} \\
        &= \lim_{x \to +\infty} \frac{0}{x + 1 + \sqrt{x^2 + 2x + 1}} \\
        &= 0
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to +\infty} (x + 1 - \sqrt{x^2 + 2x + 1}) = 0}

    \vspace{0.6cm}

    \item 
    $
    \displaystyle \lim_{x \to +\infty} \sqrt{x^2 + 1} - \sqrt{4x^2 + x - 1}:
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} \sqrt{x^2 + 1} = +\infty\\[3mm]
    &\textit{donc la somme est une forme indéterminée.}\\
    \displaystyle \lim_{x \to +\infty} -\sqrt{4x^2 + x - 1} = -\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to +\infty} \sqrt{x^2 + 1} - \sqrt{4x^2 + x - 1} &= \lim_{x \to +\infty} |x|\sqrt{1 + \dfrac{1}{x^2}} - |x|\sqrt{4 + \dfrac{1}{x} - \dfrac{1}{x^2}} \\
        &= \lim_{x \to +\infty} x\sqrt{1 + \dfrac{1}{x^2}} - x\sqrt{4 + \dfrac{1}{x} - \dfrac{1}{x^2}} \quad (|x| = x \text{ car } x > 0) \\
        &= \lim_{x \to +\infty} x \left( \sqrt{1 + \dfrac{1}{x^2}} - \sqrt{4 + \dfrac{1}{x} - \dfrac{1}{x^2}} \right) \\
        &= \begin{cases}
    \displaystyle \lim_{x \to +\infty} x = +\infty \\[3mm]
    &\text{ Par produit }\\
    \displaystyle \lim_{x \to +\infty} \left( \sqrt{1 + \frac{1}{x^2}} - \sqrt{4 + \frac{1}{x} - \frac{1}{x^2}} \right) = 1 - 2 = -1
    \end{cases}
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to +\infty} (\sqrt{x^2 + 1} - \sqrt{4x^2 + x - 1}) = -\infty}

    \vspace{0.6cm}

    \item 
    $
    \displaystyle \lim_{x \to +\infty} \sqrt{x^2 + 1} + \sqrt{x^2 + 2x + 3}
    $

    \textit{Calcul direct (pas de forme indéterminée) :}
    
    \textit{On a :}
    \[
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} \sqrt{x^2 + 1} = +\infty \\[3mm]
    &\text{ Par somme }\\
    \displaystyle \lim_{x \to +\infty} \sqrt{x^2 + 2x + 3} = +\infty
    \end{cases}
    \]
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to +\infty} (\sqrt{x^2 + 1} + \sqrt{x^2 + 2x + 3}) = +\infty}
\end{enumerate}

\subsection*{\textcolor{darkred}{\underline{5. Limites à l'infini des fonctions du type : $f(x) = \frac{u(x)}{v(x)}$,} \\[0.15cm] \underline{où $u$ et $v$ comportent des radicaux}}}

\subsubsection*{\underline{\textcolor{darkred}{ Méthode : }}}

Soit $f$ une fonction du type : $\dfrac{u(x)}{v(x)}$, avec $u$ et $v$ des fonctions comportant des radicaux.
\begin{itemize}
    \item Si le calcul direct de la limite de $f$ donne une forme indéterminée, alors on peut chercher à factoriser $x$ au numérateur et au dénominateur, puis simplifier $f(x)$.\\
    Ensuite, on recalcule la limite avec l'expression simplifiée de $f(x)$.
    \item Si on voit qu'en mettant $x$ en facteur, le calcul de la limite du quotient donnerait une forme indéterminée, alors on peut réécrire $f(x)$ en utilisant les expressions conjuguées de $u(x)$ et $v(x)$ avant de mettre $x$ en facteur au numérateur et au dénominateur et de recalculer la limite du quotient obtenu.
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple : }}}
\begin{enumerate}
\item \textit{Soit $g$ la fonction définie par $g(x) = \dfrac{x + \sqrt{x^2 + 4}}{2x + \sqrt{x^2 + 1}}$. Calculons $\displaystyle \lim_{x \to +\infty} g(x)$.}
    
    \vspace{0.1cm}
    
    $
    \displaystyle \lim_{x \to +\infty} g(x):
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} (x + \sqrt{x^2 + 4}) = +\infty\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to +\infty} (2x + \sqrt{x^2 + 1}) = +\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to +\infty} \frac{x + \sqrt{x^2 + 4}}{2x + \sqrt{x^2 + 1}} &= \lim_{x \to +\infty} \frac{x + |x|\sqrt{1 + \dfrac{4}{x^2}}}{2x + |x|\sqrt{1 + \dfrac{1}{x^2}}} \\
        &= \lim_{x \to +\infty} \frac{x + x\sqrt{1 + \dfrac{4}{x^2}}}{2x + x\sqrt{1 + \dfrac{1}{x^2}}} \quad (|x| = x \text{ car } x > 0) \\
        &= \lim_{x \to +\infty} \frac{x \left( 1 + \sqrt{1 + \dfrac{4}{x^2}} \right)}{x \left( 2 + \sqrt{1 + \dfrac{1}{x^2}} \right)} \\
        &= \lim_{x \to +\infty} \frac{1 + \sqrt{1 + \dfrac{4}{x^2}}}{2 + \sqrt{1 + \dfrac{1}{x^2}}} \\
        &= \begin{cases}
    \displaystyle \lim_{x \to +\infty} \left( 1 + \sqrt{1 + \frac{4}{x^2}} \right) = 2 \\[3mm]
    &\text{ Par quotient }\\
    \displaystyle \lim_{x \to +\infty} \left( 2 + \sqrt{1 + \frac{1}{x^2}} \right) = 3
    \end{cases}
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to +\infty} g(x) = \frac{2}{3}}
    
    \item \textit{Soit $f$ la fonction définie par $f(x) = \dfrac{x - \sqrt{x^2 + x + 1}}{2x - \sqrt{4x^2 + 3x + 1}}$. Calculons $\displaystyle \lim_{x \to +\infty} f(x)$.}
    
    \vspace{0.1cm}
    
    $
    \displaystyle \lim_{x \to +\infty} f(x):
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} (x - \sqrt{x^2 + x + 1}) = \text{F.I.}\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to +\infty} (2x - \sqrt{4x^2 + 3x + 1}) = \text{F.I.}
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to +\infty} \frac{x - \sqrt{x^2 + x + 1}}{2x - \sqrt{4x^2 + 3x + 1}} &= \lim_{x \to +\infty} \frac{\left(x - \sqrt{x^2 + x + 1}\right)\left(x + \sqrt{x^2 + x + 1}\right)\left(2x + \sqrt{4x^2 + 3x + 1}\right)}{\left(2x - \sqrt{4x^2 + 3x + 1}\right)\left(2x + \sqrt{4x^2 + 3x + 1}\right)\left(x + \sqrt{x^2 + x + 1}\right)} \\
        &= \lim_{x \to +\infty} \frac{\left[x^2 - (x^2 + x + 1)\right]\left(2x + \sqrt{4x^2 + 3x + 1}\right)}{\left[4x^2 - (4x^2 + 3x + 1)\right]\left(x + \sqrt{x^2 + x + 1}\right)} \\
        &= \lim_{x \to +\infty} \frac{(-x - 1)\left(2x + \sqrt{4x^2 + 3x + 1}\right)}{(-3x - 1)\left(x + \sqrt{x^2 + x + 1}\right)} \\
        &= \lim_{x \to +\infty} \frac{x\left(-1 - \dfrac{1}{x}\right) \cdot x\left(2 + \sqrt{4 + \dfrac{3}{x} + \dfrac{1}{x^2}}\right)}{x\left(-3 - \dfrac{1}{x}\right) \cdot x\left(1 + \sqrt{1 + \dfrac{1}{x} + \dfrac{1}{x^2}}\right)} \quad (|x| = x \text{ car } x > 0) \\
        &= \lim_{x \to +\infty} \frac{x^2\left(-1 - \dfrac{1}{x}\right)\left(2 + \sqrt{4 + \dfrac{3}{x} + \dfrac{1}{x^2}}\right)}{x^2\left(-3 - \dfrac{1}{x}\right)\left(1 + \sqrt{1 + \dfrac{1}{x} + \dfrac{1}{x^2}}\right)} \\
        &= \lim_{x \to +\infty} \frac{\left(-1 - \dfrac{1}{x}\right)\left(2 + \sqrt{4 + \dfrac{3}{x} + \dfrac{1}{x^2}}\right)}{\left(-3 - \dfrac{1}{x}\right)\left(1 + \sqrt{1 + \dfrac{1}{x} + \dfrac{1}{x^2}}\right)} \\
        &= \begin{cases}
    \displaystyle \lim_{x \to +\infty} \left(-1 - \frac{1}{x}\right)\left(2 + \sqrt{4 + \frac{3}{x} + \frac{1}{x^2}}\right) = (-1)(2 + 2) = -4 \\[3mm]
    &\text{ Par quotient }\\
    \displaystyle \lim_{x \to +\infty} \left(-3 - \frac{1}{x}\right)\left(1 + \sqrt{1 + \frac{1}{x} + \frac{1}{x^2}}\right) = (-3)(1 + 1) = -6
    \end{cases}
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to +\infty} f(x) = \frac{2}{3}}
\end{enumerate}

\subsubsection*{\underline{\textcolor{darkred}{ Exercice : }}}

Calculer les limites suivantes :


\begin{enumerate}
    \item 
    $
    \displaystyle \lim_{x \to -\infty} \frac{x + \sqrt{4x^2 + 1}}{x - 1}:
    \begin{cases}
    \displaystyle \lim_{x \to -\infty} (x + \sqrt{4x^2 + 1}) = \text{F.I.}\\[3mm]
    &\textit{donc le quotient est une forme indéterminée.}\\
    \displaystyle \lim_{x \to -\infty} (x - 1) = -\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to -\infty} \frac{x + \sqrt{4x^2 + 1}}{x - 1} &= \lim_{x \to -\infty} \frac{x + |x|\sqrt{4 + \dfrac{1}{x^2}}}{x\left(1 - \dfrac{1}{x}\right)} \\
        &= \lim_{x \to -\infty} \frac{x - x\sqrt{4 + \dfrac{1}{x^2}}}{x\left(1 - \dfrac{1}{x}\right)} \quad (|x| = -x \text{ car } x < 0) \\
        &= \lim_{x \to -\infty} \frac{x\left(1 - \sqrt{4 + \dfrac{1}{x^2}}\right)}{x\left(1 - \dfrac{1}{x}\right)} \\
        &= \lim_{x \to -\infty} \frac{1 - \sqrt{4 + \dfrac{1}{x^2}}}{1 - \dfrac{1}{x}} \\
        &= \begin{cases}
    \displaystyle \lim_{x \to -\infty} \left(1 - \sqrt{4 + \frac{1}{x^2}}\right) = 1 - 2 = -1 \\[3mm]
    &\text{ Par quotient }\\
    \displaystyle \lim_{x \to -\infty} \left(1 - \frac{1}{x}\right) = 1
    \end{cases}
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to -\infty} \frac{x + \sqrt{4x^2 + 1}}{x - 1} = -1}

    \vspace{0.6cm}

    \item 
    $
    \displaystyle \lim_{x \to +\infty} \frac{x + 1 - \sqrt{x^2 + 2x + 1}}{x}:
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} (x + 1 - \sqrt{x^2 + 2x + 1}) = \text{F.I.}\\[3mm]
    &\textit{donc le quotient est une FI.}\\
    \displaystyle \lim_{x \to +\infty} x = +\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to +\infty} \frac{x + 1 - \sqrt{x^2 + 2x + 1}}{x} &= \lim_{x \to +\infty} \frac{(x + 1 - \sqrt{x^2 + 2x + 1})(x + 1 + \sqrt{x^2 + 2x + 1})}{x(x + 1 + \sqrt{x^2 + 2x + 1})} \\
        &= \lim_{x \to +\infty} \frac{(x + 1)^2 - (x^2 + 2x + 1)}{x(x + 1 + \sqrt{x^2 + 2x + 1})} \\
        &= \lim_{x \to +\infty} \frac{0}{x(x + 1 + \sqrt{x^2 + 2x + 1})} \\
        &= 0
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to +\infty} \frac{x + 1 - \sqrt{x^2 + 2x + 1}}{x} = 0}

    \vspace{0.6cm}

    \item 
    $
    \displaystyle \lim_{x \to +\infty} \frac{\sqrt{x^2 + 1} - \sqrt{4x^2 + x - 1}}{x - \sqrt{4x^2 - 1}}:
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} (\sqrt{x^2 + 1} - \sqrt{4x^2 + x - 1}) = -\infty\\[3mm]
    &\textit{donc le quotient est une FI.}\\
    \displaystyle \lim_{x \to +\infty} (x - \sqrt{4x^2 - 1}) = -\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to +\infty} \frac{\sqrt{x^2 + 1} - \sqrt{4x^2 + x - 1}}{x - \sqrt{4x^2 - 1}} &= \lim_{x \to +\infty} \frac{|x|\sqrt{1 + \dfrac{1}{x^2}} - |x|\sqrt{4 + \dfrac{1}{x} - \dfrac{1}{x^2}}}{x - |x|\sqrt{4 - \dfrac{1}{x^2}}} \\
        &= \lim_{x \to +\infty} \frac{x\sqrt{1 + \dfrac{1}{x^2}} - x\sqrt{4 + \dfrac{1}{x} - \dfrac{1}{x^2}}}{x - x\sqrt{4 - \dfrac{1}{x^2}}} \quad (|x| = x \text{ car } x > 0) \\
        &= \lim_{x \to +\infty} \frac{x\left(\sqrt{1 + \dfrac{1}{x^2}} - \sqrt{4 + \dfrac{1}{x} - \dfrac{1}{x^2}}\right)}{x\left(1 - \sqrt{4 - \dfrac{1}{x^2}}\right)} \\
        &= \lim_{x \to +\infty} \frac{\sqrt{1 + \dfrac{1}{x^2}} - \sqrt{4 + \dfrac{1}{x} - \dfrac{1}{x^2}}}{1 - \sqrt{4 - \dfrac{1}{x^2}}} \\
        &= \begin{cases}
    \displaystyle \lim_{x \to +\infty} \left(\sqrt{1 + \frac{1}{x^2}} - \sqrt{4 + \frac{1}{x} - \frac{1}{x^2}}\right) = 1 - 2 = -1 \\[3mm]
    &\text{ Par quotient }\\
    \displaystyle \lim_{x \to +\infty} \left(1 - \sqrt{4 - \frac{1}{x^2}}\right) = 1 - 2 = -1
    \end{cases}
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to +\infty} \frac{\sqrt{x^2 + 1} - \sqrt{4x^2 + x - 1}}{x - \sqrt{4x^2 - 1}} = 1}

    \vspace{0.6cm}

    \item 
    $
    \displaystyle \lim_{x \to +\infty} \frac{\sqrt{x^2 + 1} + \sqrt{x^2 + 2x + 3}}{\sqrt{x^2 - 1} + \sqrt{x^2 + x - 1}}:
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} (\sqrt{x^2 + 1} + \sqrt{x^2 + 2x + 3}) = +\infty\\[3mm]
    &\textit{donc le quotient est une FI.}\\
    \displaystyle \lim_{x \to +\infty} (\sqrt{x^2 - 1} + \sqrt{x^2 + x - 1}) = +\infty
    \end{cases}
    $

    \textit{Levons l'indétermination :}
    
    \textit{On a :}
    \begin{align*}
        \displaystyle \lim_{x \to +\infty} \frac{\sqrt{x^2 + 1} + \sqrt{x^2 + 2x + 3}}{\sqrt{x^2 - 1} + \sqrt{x^2 + x - 1}} &= \lim_{x \to +\infty} \frac{|x|\sqrt{1 + \dfrac{1}{x^2}} + |x|\sqrt{1 + \dfrac{2}{x} + \dfrac{3}{x^2}}}{|x|\sqrt{1 - \dfrac{1}{x^2}} + |x|\sqrt{1 + \dfrac{1}{x} - \dfrac{1}{x^2}}} \\
        &= \lim_{x \to +\infty} \frac{x\sqrt{1 + \dfrac{1}{x^2}} + x\sqrt{1 + \dfrac{2}{x} + \dfrac{3}{x^2}}}{x\sqrt{1 - \dfrac{1}{x^2}} + x\sqrt{1 + \dfrac{1}{x} - \dfrac{1}{x^2}}} \quad (|x| = x \text{ car } x > 0) \\
        &= \lim_{x \to +\infty} \frac{x\left(\sqrt{1 + \dfrac{1}{x^2}} + \sqrt{1 + \dfrac{2}{x} + \dfrac{3}{x^2}}\right)}{x\left(\sqrt{1 - \dfrac{1}{x^2}} + \sqrt{1 + \dfrac{1}{x} - \dfrac{1}{x^2}}\right)} \\
        &= \lim_{x \to +\infty} \frac{\sqrt{1 + \dfrac{1}{x^2}} + \sqrt{1 + \dfrac{2}{x} + \dfrac{3}{x^2}}}{\sqrt{1 - \dfrac{1}{x^2}} + \sqrt{1 + \dfrac{1}{x} - \dfrac{1}{x^2}}} \\
        &= \begin{cases}
    \displaystyle \lim_{x \to +\infty} \left(\sqrt{1 + \frac{1}{x^2}} + \sqrt{1 + \frac{2}{x} + \frac{3}{x^2}}\right) = 1 + 1 = 2 \\[3mm]
    &\text{ Par quotient }\\
    \displaystyle \lim_{x \to +\infty} \left(\sqrt{1 - \frac{1}{x^2}} + \sqrt{1 + \frac{1}{x} - \frac{1}{x^2}}\right) = 1 + 1 = 2
    \end{cases}
    \end{align*}
    
    \vspace{0.2cm}
    
    \text{\textit{Donc, }} \boxed{\lim_{x \to +\infty} \frac{\sqrt{x^2 + 1} + \sqrt{x^2 + 2x + 3}}{\sqrt{x^2 - 1} + \sqrt{x^2 + x - 1}} = 1}
\end{enumerate}
\subsection*{\textcolor{darkred}{\underline{6. Limite et comparaison de fonction}}}
\subsubsection*{\underline{\textcolor{darkred}{ Propriété : (Théorème des gendarmes) }}}
Soient $f$, $g$ et $h$ des fonctions définies sur un intervalle ouvert $I$ et $x_0 \in I$.

Si pour tout $x \in I$, on a : $g(x) \le f(x) \le h(x)$ et que $\displaystyle \lim_{x \to x_0} g(x) = \lim_{x \to x_0} h(x) = l$, alors $\displaystyle \lim_{x \to x_0} f(x) = l$

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}
\begin{enumerate}
    \item \textit{Soit $f(x) = \dfrac{\sin 3x + 4}{x}$. Calculons : $\displaystyle \lim_{x \to +\infty} f(x)$.}
    
    \textit{Soit $x \in ]0 \,;\, +\infty[$, on a :}
    \begin{align*}
        -1 \le \sin 3x \le 1 &\implies 3 \le \sin 3x + 4 \le 5 \implies \frac{3}{x} \le \frac{\sin 3x + 4}{x} \le \frac{5}{x} \\
        &\implies \frac{3}{x} \le f(x) \le \frac{5}{x}.
    \end{align*}

    \textit{Or, } $\displaystyle \lim_{x \to +\infty} \frac{3}{x} = \lim_{x \to +\infty} \frac{5}{x} = 0$.

    \textit{Donc d'après le théorème des gendarmes, } $\displaystyle \lim_{x \to +\infty} f(x) = 0$

    \vspace{0.6cm}

    \item \textit{Soit $g(x) = \dfrac{E(x)}{x}$. Calculons : $\displaystyle \lim_{x \to +\infty} g(x)$.}

    \textit{Soit $x \in ]0 \,;\, +\infty[$, on a :}
    \begin{align*}
        E(x) \le x < E(x) + 1 &\implies x - 1 < E(x) \le x \implies \frac{x - 1}{x} < \frac{E(x)}{x} \le 1 \\
        &\implies \frac{x - 1}{x} < g(x) \le 1.
    \end{align*}

    \textit{Or, } $\displaystyle \lim_{x \to +\infty} \frac{x - 1}{x} = 1$. \textit{Donc d'après le théorème de gendarme } $\displaystyle \lim_{x \to +\infty} g(x) = 1$.
\end{enumerate}

\subsubsection*{\underline{\textcolor{darkred}{ Propriété : (Théorème de majoration) }}}

Soient $f$ et $g$ deux fonctions définies sur un intervalle ouvert $I$ et $x_0 \in I$.

Si pour tout $x \in I$, on a : $|f(x) - l| \le g(x)$ et $\displaystyle \lim_{x \to x_0} g(x) = 0$, alors $\displaystyle \lim_{x \to x_0} f(x) = l$

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}

Soit $f(x) = \dfrac{\cos 2x}{x^2}$. Calculons : $\displaystyle \lim_{x \to +\infty} f(x)$.
\vspace{0.2cm}

\textit{Soit $x \in ]0 \,;\, +\infty[$, on a :}
\begin{align*}
    |\cos 2x| \le 1 &\implies \frac{|\cos 2x|}{x^2} \le \frac{1}{x^2} \implies \left|\frac{\cos 2x}{x^2}\right| \le \frac{1}{x^2} \implies |f(x)| \le \frac{1}{x^2}.
\end{align*}

\vspace{0.1cm}

\textit{Or, } $\displaystyle \lim_{x \to +\infty} \frac{1}{x^2} = 0$. \textit{Donc d'après le théorème des majoration, } $\displaystyle \lim_{x \to +\infty} f(x) = 0$

\subsection*{\textcolor{darkred}{\underline{6. Limite et changement de variable}}}
\subsubsection*{\underline{\textcolor{darkred}{ Propriété : }}}

Soit $f$ une fonction et $x_0 \in \mathbb{R}$.

\begin{itemize}
    \item En posant $X = x - x_0$, on a :  $\displaystyle \lim_{x \to x_0} f(x) = \lim_{X \to 0} f(X + x_0)$.
    \item En posant $X = \dfrac{1}{x}$, on a :  $\displaystyle \lim_{x \to x_0} f(x) = \lim_{X \to 0} f\left(\frac{1}{X}\right)$.
\end{itemize}

En effet,

\begin{itemize}
    \item En faisant le changement de variable $X = x - x_0$, on a : 
    $\begin{cases}
        x \to x_0 \iff X \to 0 \\
        X = x - x_0 \iff x = X + x_0
    \end{cases}.$ \\
    Ainsi : $\displaystyle \lim_{x \to x_0} f(x) = \lim_{X \to 0} f(X + x_0)$.

    \vspace{0.3cm}

    \item En faisant le changement de variable $X = \dfrac{1}{x}$, on a : 
    $\begin{cases}
        x \to +\infty \iff X \to 0 \\
        X = \dfrac{1}{x} \iff x = \dfrac{1}{X}
    \end{cases}.$ \\
    Donc : $\displaystyle \lim_{x \to +\infty} f(x) = \lim_{X \to 0} f\left(\frac{1}{X}\right)$.
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}

\vspace{0.2cm}

\begin{enumerate}
    \item \textit{Calculons : } $\displaystyle \lim_{x \to 1} \frac{1}{\sqrt{x - 1}}$.
    
    \vspace{0.1cm}
    
    \textit{En posant $X = x - 1$, on a : } 
    $\displaystyle \lim_{x \to 1} \frac{1}{\sqrt{x - 1}} = \lim_{X \to 0} \frac{1}{\sqrt{X}} = \ll \frac{1}{0^+} \gg = +\infty$.

    \vspace{0.2cm}

    \hfil \boxed{\lim_{x \to 1} \frac{1}{\sqrt{x - 1}} = +\infty} \hfil

    \vspace{0.6cm}

    \item \textit{Calculons : } $\displaystyle \lim_{x \to +\infty} \frac{1}{x} E\left(\frac{1}{x}\right)$.

    \vspace{0.1cm}

    \textit{En posant $X = \dfrac{1}{x}$, on a : } 
    $\displaystyle \lim_{x \to +\infty} \frac{1}{x} E\left(\frac{1}{x}\right) = \lim_{X \to 0} X E(X) = 0 \times E(0) = 0$.

    \vspace{0.2cm}

    \hfil \boxed{\lim_{x \to +\infty} \frac{1}{x} E\left(\frac{1}{x}\right) = 0} \hfil
\end{enumerate}

\section*{\underline{\textcolor{darkred}{ IV. Continuité d'une fonction }}}
\subsection*{\underline{\textcolor{darkred}{ 1. Notion intuitive de continuité }}}

Intuitivement, une fonction $f$ est continue sur un intervalle si l'on peut tracer sa représentation graphique « sans lever le crayon », c'est-à-dire sans saut ni interruption.

\subsubsection*{Activité}
Soient les fonctions $f$ et $g$ définies sur $\mathbb{R}$ par :
\[
f(x) = x^2 \quad \text{et} \quad g(x) = \begin{cases} x + 1 & \text{si } x \ge 0 \\ x - 1 & \text{si } x < 0 \end{cases}
\]
\begin{enumerate}[label=\arabic*.]
    \item Calculer $\displaystyle\lim_{x \to 0} f(x)$ et $f(0)$.
    \item Calculer $\displaystyle\lim_{x \to 0^+} g(x)$, $\displaystyle\lim_{x \to 0^-} g(x)$ et $g(0)$.
    \item Que remarque-t-on graphiquement au point d'abscisse $x_0 = 0$ pour chacune de ces fonctions ?
\end{enumerate}

\subsubsection*{Interprétation graphique}
\begin{itemize}
    \item Pour $f$, la courbe $(\mathcal{C}_f)$ ne présente aucune rupture en $x_0 = 0$.
    \item Pour $g$, la courbe $(\mathcal{C}_g)$ subit un « saut » en $x_0 = 0$ : il y a une interruption du tracé.
\end{itemize}

\subsubsection*{Exemple}
La fonction $h : x \mapsto |x|$ est continue sur $\mathbb{R}$. Son graphe est formé de deux demi-droites qui se rejoignent en l'origine $(0,0)$ sans rupture.

\vspace{0.5cm}
\subsection{Continuité en un point}

\begin{tcolorbox}[
    colframe=primary,
    colback=boxbg,
    coltitle=white,
    colbacktitle=titlebg,
    fonttitle=\bfseries,
    title=Définition,
    arc=3mm,
    boxrule=1pt
]
Soit $f$ une fonction définie sur un ensemble $D_f$ et $x_0 \in D_f$.\\
On dit que $f$ est \textbf{continue en $x_0$} si :
\[
\lim_{x \to x_0} f(x) = f(x_0)
\]
\end{tcolorbox}

\noindent\textbf{Remarque :} Cela implique que $f$ doit être définie en $x_0$ et que les limites à gauche et à droite en $x_0$ existent et sont égales à $f(x_0)$ :
\[
\lim_{x \to x_0^-} f(x) = \lim_{x \to x_0^+} f(x) = f(x_0)
\]

\subsubsection*{Interprétation}
Lorsque $x$ se rapproche de $x_0$, l'image $f(x)$ se rapproche de $f(x_0)$. Le point de la courbe de coordonnées $(x_0, f(x_0))$ raccordement parfaitement les parties gauche et droite du graphe.

\subsubsection*{Exemple}
Soit la fonction $f$ définie par $f(x) = \dfrac{x^2 - 1}{x - 1}$ pour $x \neq 1$ et $f(1) = 2$.
\begin{itemize}
    \item Pour $x \neq 1$, $f(x) = \dfrac{(x-1)(x+1)}{x-1} = x + 1$.
    \item $\displaystyle\lim_{x \to 1} f(x) = \lim_{x \to 1} (x+1) = 2$.
    \item Comme $\displaystyle\lim_{x \to 1} f(x) = f(1) = 2$, la fonction $f$ est continue en $1$.
\end{itemize}

\subsubsection*{Contre-exemple}
Soit la fonction partie entière $f(x) = E(x)$ au point $x_0 = 2$.
\begin{itemize}
    \item $f(2) = 2$
    \item $\displaystyle\lim_{x \to 2^+} E(x) = 2$ \quad et \quad $\displaystyle\lim_{x \to 2^-} E(x) = 1$
\end{itemize}
Puisque $\displaystyle\lim_{x \to 2^-} E(x) \neq \lim_{x \to 2^+} E(x)$, $f$ n'admet pas de limite en $2$. La fonction $E$ n'est pas continue en $2$.

\subsubsection*{Exercice}
Soit $f$ la fonction définie sur $\mathbb{R}$ par :
\[
f(x) = \begin{cases} \dfrac{x^2 - 4}{x - 2} & \text{si } x \neq 2 \\[8pt] k & \text{si } x = 2 \end{cases}
\]
Déterminer la valeur du réel $k$ pour que $f$ soit continue en $x_0 = 2$.

\vspace{0.5cm}
\subsection{Continuité sur un intervalle}

\begin{tcolorbox}[
    colframe=primary,
    colback=boxbg,
    coltitle=white,
    colbacktitle=titlebg,
    fonttitle=\bfseries,
    title=Définition,
    arc=3mm,
    boxrule=1pt
]
Soit $I$ un intervalle de $\mathbb{R}$.\\
Une fonction $f$ est dite \textbf{continue sur l'intervalle $I$} si elle est continue en tout point de $I$.
\end{tcolorbox}

\subsubsection*{Interprétation graphique}
La courbe représentative de $f$ sur l'intervalle $I$ se trace d'un seul tenant, sans lever le crayon.

\subsubsection*{Exemple}
La fonction $f : x \mapsto \sqrt{x}$ est continue sur $[0, +\infty[$.

\subsubsection*{Exercice}
Justifier la continuité de la fonction $f(x) = 2x^3 - 5x + 1$ sur $\mathbb{R}$.

\vspace{0.5cm}
\subsection{Opérations sur les fonctions continues}

\begin{tcolorbox}[
    colframe=primary,
    colback=boxbg,
    coltitle=white,
    colbacktitle=titlebg,
    fonttitle=\bfseries,
    title=Propriété,
    arc=3mm,
    boxrule=1pt
]
Soient $f$ et $g$ deux fonctions continues sur un intervalle $I$, et $k \in \mathbb{R}$.
\begin{enumerate}[label=\arabic*.]
    \item Les fonctions $f + g$, $f \times g$ et $k \cdot f$ sont continues sur $I$.
    \item Si $g$ ne s'annule pas sur $I$ ($g(x) \neq 0$), alors $\dfrac{1}{g}$ et $\dfrac{f}{g}$ sont continues sur $I$.
    \item Si $f$ est strictement positive sur $I$, alors $\sqrt{f}$ est continue sur $I$.
\end{enumerate}
\end{tcolorbox}

\subsubsection*{Exemple}
La fonction $h(x) = \dfrac{x^2 + 3}{x - 1}$ est continue sur $]-\infty, 1[$ et sur $]1, +\infty[$ comme quotient de deux fonctions polynômes continues dont le dénominateur ne s'annule pas.

\subsubsection*{Exercice}
Étudier la continuité de $f(x) = \sqrt{x^2 - 4}$ sur son domaine de définition.

\vspace{0.5cm}
\subsection{Fonctions usuelles continues}

\begin{tcolorbox}[
    colframe=primary,
    colback=boxbg,
    coltitle=white,
    colbacktitle=titlebg,
    fonttitle=\bfseries,
    title=Propriété,
    arc=3mm,
    boxrule=1pt
]
\begin{itemize}
    \item Toute fonction \textbf{polynôme} est continue sur $\mathbb{R}$.
    \item Toute fonction \textbf{rationnelle} (quotient de deux polynômes) est continue sur tout intervalle contenu dans son ensemble de définition.
    \item La fonction \textbf{racine carrée} $x \mapsto \sqrt{x}$ est continue sur $[0, +\infty[$.
    \item Les fonctions \textbf{valeur absolue} ($x \mapsto |x|$), \textbf{sinus} ($x \mapsto \sin x$) et \textbf{cosinus} ($x \mapsto \cos x$) sont continues sur $\mathbb{R}$.
\end{itemize}
\end{tcolorbox}

\subsubsection*{Exemple}
$f(x) = 3x^4 - 2x + 7$ est une fonction polynôme, elle est donc continue sur $\mathbb{R}$.

\subsubsection*{Exercice}
Déterminer les intervalles de continuité de la fonction $g(x) = \dfrac{\cos(x)}{x^2 - 9}$.

\vspace{0.5cm}
\subsection{Théorème des valeurs intermédiaires (TVI)}

\begin{tcolorbox}[
    colframe=primary,
    colback=boxbg,
    coltitle=white,
    colbacktitle=titlebg,
    fonttitle=\bfseries,
    title=Énoncé du TVI,
    arc=3mm,
    boxrule=1pt
]
Soit $f$ une fonction continue sur un intervalle $[a, b]$.\\
Pour tout réel $k$ compris entre $f(a)$ et $f(b)$, il existe au moins un réel $c \in [a, b]$ tel que :
\[
f(c) = k
\]
\end{tcolorbox}

\noindent\textbf{Corollaire (Existence d'une racine) :} Si $f$ est continue sur $[a, b]$ et si $f(a) \times f(b) < 0$, alors l'équation $f(x) = 0$ admet au moins une solution $c \in ]a, b[$.

\subsubsection*{Interprétation graphique}
La courbe $(\mathcal{C}_f)$ reliant le point $(a, f(a))$ au point $(b, f(b))$ sans rupture coupe obligatoirement la droite horizontale d'équation $y = k$ en au moins un point.

\subsubsection*{Exemple}
Soit $f(x) = x^3 + x - 1$ sur $[0, 1]$.
\begin{itemize}
    \item $f$ est continue sur $[0, 1]$ (fonction polynôme).
    \item $f(0) = -1$ et $f(1) = 1$.
    \item $0$ est compris entre $f(0)$ et $f(1)$ (car $f(0) \times f(1) < 0$).
\end{itemize}
D'après le TVI, l'équation $x^3 + x - 1 = 0$ admet au moins une solution $c \in ]0, 1[$.

\subsubsection*{Application à la résolution d'équations}
Le TVI permet de prouver l'\textbf{existence} de solutions pour des équations de la forme $f(x) = k$, notamment lorsque l'équation ne peut pas être résolue explicitement par calcul algébrique direct.

\subsubsection*{Exercice}
Montrer que l'équation $x^5 - 3x - 1 = 0$ admet au moins une solution réelle dans l'intervalle $[1, 2]$.
