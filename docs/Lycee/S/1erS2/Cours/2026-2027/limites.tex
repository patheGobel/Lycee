\textit{

}

}

}

\textit{
on dit que la limite à droite de \(0\) de \(f\) est égale à \(+\infty\).

\subsubsection*{\underline{\textcolor{darkred}{ Interprétation :}}}

\begin{itemize}
\item On constate que lorsque $x$ prend des valeurs positives de plus en plus grandes, $f(x)$ prend des valeurs positives de plus en plus grandes. \\
On dit alors que la limite lorsque $x$ tend vers $+\infty$ de $f(x)$ est égale à $+\infty$. \\
On note :
\[
\boxed{\lim_{x \to +\infty} f(x) = +\infty}
\]

\item On constate que lorsque $x$ prend des valeurs positives de plus en plus grandes, $g(x)$ prend des valeurs positives de plus en plus proche de $0$. \\
On dit alors que la limite lorsque $x$ tend vers $+\infty$ de $g(x)$ est égale à $0$. \\
On note :
\[
\boxed{\lim_{x \to +\infty} g(x) = 0}
\]
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Théorème de l’unicité de la limite }}}
Soit $f$ une fonction.\\
Si $f$ admet une limite en un point $x_0$, en $+\infty$ ou en $-\infty$, alors cette limite est unique.

\subsection*{\underline{\textcolor{darkred}{ 2. Limites à l’infini des fonctions usuelles}}}
\subsubsection*{\underline{\textcolor{darkred}{ Propriété }}}

\noindent \textit{Soient $a \in \mathbb{R}$ et $n \in \mathbb{N} \setminus \{0\}$, on a :}

\vspace{0.4cm}

\begin{multicols}{2}
\noindent
1) $\displaystyle \lim_{x \to +\infty} a = a$ \\[0.5cm]
2) $\displaystyle \lim_{x \to +\infty} x^n = +\infty$ \\[0.5cm]
3) $\displaystyle \lim_{x \to +\infty} \frac{1}{x^n} = 0$

\columnbreak

\noindent
4) $\displaystyle \lim_{x \to -\infty} a = a$ \\[0.3cm]
5) $\displaystyle \lim_{x \to -\infty} x^n = \begin{cases} +\infty & \text{\textit{si $n$ est pair}} \\ -\infty & \text{\textit{si $n$ est impair}} \end{cases}$ \\[0.3cm]
6) $\displaystyle \lim_{x \to -\infty} \frac{1}{x^n} = 0$
\end{multicols}

\begin{multicols}{3}

\end{itemize}

\subsection*{\underline{\textcolor{darkred}{ 5. Limite à gauche et à droite en un point $x_{0}$ }}}
\subsubsection*{\underline{\textcolor{darkred}{ a. Définition }}}

\textit{Soit $f$ une fonction et $x_0 \in \mathbb{R}$.}

\begin{itemize}
\item \textit{On appelle limite à gauche en $x_0$ de $f$, la limite de $f$ lorsque $x$ tend vers $x_0$, en restant à gauche de $x_0$. On la note :}
\[
\lim_{x \to x_0^-} f(x) \quad \text{\textit{ou}} \quad \lim_{\substack{x \to x_0 \\ x < x_0}} f(x)
\]

\vspace{0.3cm}
\item \textit{On appelle limite à droite en $x_0$ de $f$, la limite de $f$ lorsque $x$ tend vers $x_0$, en restant à droite de $x_0$. On la note :}
\[
\lim_{x \to x_0^+} f(x) \quad \text{\textit{ou}} \quad \lim_{\substack{x \to x_0 \\ x > x_0}} f(x)
\]

\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}

Soit $f$ la fonction définie par : 
$
f(x) =
\begin{cases} 
2x - 1 & \text{\textit{si }} x > 1 \\ 
x^2 + x + 1 & \text{\textit{si }} x \leq 1 
\end{cases}
$

\vspace{0.3cm}

\noindent \textit{Calculons les limites en $1^-$ et en $1^+$ de $f$.} \\
\textit{On a :}

\begin{itemize}
    \item .
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 1^-} f(x) &= \lim_{x \to 1^-} x^2 + x + 1 \\
    &= (1)^2 + (1) + 1 \\
    &= 3
    \end{aligned}
    $
    
    $\boxed{\lim_{x \to 1^-} f(x) = 3}$
    
    \vspace{0.3cm}
    \item .
    
     $
    \begin{aligned}
    \displaystyle \lim_{x \to 1^+} f(x) &= \lim_{x \to 1^+} 2x - 1 \\
    &= 2(1) - 1 \\
    &= 1
    \end{aligned}
    $
    
    $ \boxed{\lim_{x \to 1^+} f(x) = 1}$
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ b. Propriété }}}

Pour tout réel $a$, on a :
\[
\lim_{x \to a^+} \frac{1}{x - a} = +\infty \quad ; \quad \lim_{x \to a^-} \frac{1}{x - a} = -\infty
\]

\begin{center}
\begin{tikzpicture}[node style/.style={fill opacity=0,text opacity=1}]
        \tkzTabInit[espcl=2,lgt=3]{$x$/.5,$x - a$/.7}{$-\infty$,$a$ ,$+\infty$}
        \tkzTabLine{ ,-,z,+, }
\end{tikzpicture}
\end{center}

\paragraph{\underline{Explication :}}
Le tableau de signes de l'expression $x - a$ permet de déterminer si le dénominateur tend vers $0$ par valeurs positives ($0^+$) ou par valeurs négatives ($0^-$) :

\begin{itemize}
    \item \textbf{Limite à droite ($x \to a^+$) :} 
    Lorsque $x > a$ (à droite de $a$), le tableau indique que $x - a$ est \textbf{positif} ($+$). Le dénominateur tend donc vers $0$ par valeurs supérieures, soit $0^+$.
    \[
    \text{Par règle des signes : } \lim_{x \to a^+} \frac{1}{x - a} = \frac{1}{0^+} = +\infty
    \]

    \item \textbf{Limite à gauche ($x \to a^-$) :} 
    Lorsque $x < a$ (à gauche de $a$), le tableau indique que $x - a$ est \textbf{négatif} ($-$). Le dénominateur tend donc vers $0$ par valeurs inférieures, soit $0^-$.
    \[
    \text{Par règle des signes : } \lim_{x \to a^-} \frac{1}{x - a} = \frac{1}{0^-} = -\infty
    \]
\end{itemize}
\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}    

\noindent
\begin{multicols}{2}

\noindent \textbf{1. Limites de $\dfrac{1}{x}$ en $0$ :}

\begin{center}
\begin{tikzpicture}[node style/.style={fill opacity=0,text opacity=1}]
    \tkzTabInit[espcl=1.5,lgt=2]{$x$/.5,$x$/.7}{$-\infty$,$0$,$+\infty$}
    \tkzTabLine{ ,-,z,+, }
\end{tikzpicture}
\end{center}

\begin{itemize}
    \item \textbf{À droite ($x \to 0^+$) :} d'après le tableau, pour $x > 0$, le dénominateur est positif ($+$). Donc $x \to 0^+$.
    \[
    \lim_{x \to 0^+} \frac{1}{x} = \frac{1}{0^+} = +\infty
    \]
    \item \textbf{À gauche ($x \to 0^-$) :} d'après le tableau, pour $x < 0$, le dénominateur est négatif ($-$). Donc $x \to 0^-$.
    \[
    \lim_{x \to 0^-} \frac{1}{x} = \frac{1}{0^-} = -\infty
    \]
\end{itemize}

\columnbreak

\noindent \textbf{2. Limites de $\dfrac{1}{x - 2}$ en $2$ :}

\begin{center}
\begin{tikzpicture}[node style/.style={fill opacity=0,text opacity=1}]
    \tkzTabInit[espcl=1.5,lgt=2]{$x$/.5,$x - 2$/.7}{$-\infty$,$2$,$+\infty$}
    \tkzTabLine{ ,-,z,+, }

\end{center}

\begin{itemize}
    \item \textbf{À droite ($x \to 2^+$) :} d'après le tableau, pour $x > 2$, l'expression $x - 2$ est positive ($+$). Le dénominateur tend vers $0^+$.
    \[
    \lim_{x \to 2^+} \frac{1}{x - 2} = \frac{1}{0^+} = +\infty
    \]
    \item \textbf{À gauche ($x \to 2^-$) :} d'après le tableau, pour $x < 2$, l'expression $x - 2$ est négative ($-$). Le dénominateur tend vers $0^-$.
    \[
    \lim_{x \to 2^-} \frac{1}{x - 2} = \frac{1}{0^-} = -\infty
    \]
\end{itemize}
