\textit{

}

}

}

\textit{
on dit que la limite à droite de \(0\) de \(f\) est égale à \(+\infty\).

\subsubsection*{\underline{\textcolor{darkred}{ Interprétation :}}}

\begin{itemize}
\item On constate que lorsque $x$ prend des valeurs positives de plus en plus grandes, $f(x)$ prend des valeurs positives de plus en plus grandes. \\
On dit alors que la limite lorsque $x$ tend vers $+\infty$ de $f(x)$ est égale à $+\infty$. \\
On note :
\[
\boxed{\lim_{x \to +\infty} f(x) = +\infty}
\]

\item On constate que lorsque $x$ prend des valeurs positives de plus en plus grandes, $g(x)$ prend des valeurs positives de plus en plus proche de $0$. \\
On dit alors que la limite lorsque $x$ tend vers $+\infty$ de $g(x)$ est égale à $0$. \\
On note :
\[
\boxed{\lim_{x \to +\infty} g(x) = 0}
\]
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Théorème de l’unicité de la limite }}}
Soit $f$ une fonction.\\
Si $f$ admet une limite en un point $x_0$, en $+\infty$ ou en $-\infty$, alors cette limite est unique.

\subsection*{\underline{\textcolor{darkred}{ 2. Limites à l’infini des fonctions usuelles}}}
\subsubsection*{\underline{\textcolor{darkred}{ Propriété }}}

\noindent \textit{Soient $a \in \mathbb{R}$ et $n \in \mathbb{N} \setminus \{0\}$, on a :}

\vspace{0.4cm}

\begin{multicols}{2}
\noindent
1) $\displaystyle \lim_{x \to +\infty} a = a$ \\[0.5cm]
2) $\displaystyle \lim_{x \to +\infty} x^n = +\infty$ \\[0.5cm]
3) $\displaystyle \lim_{x \to +\infty} \frac{1}{x^n} = 0$

\columnbreak

\noindent
4) $\displaystyle \lim_{x \to -\infty} a = a$ \\[0.3cm]
5) $\displaystyle \lim_{x \to -\infty} x^n = \begin{cases} +\infty & \text{\textit{si $n$ est pair}} \\ -\infty & \text{\textit{si $n$ est impair}} \end{cases}$ \\[0.3cm]
6) $\displaystyle \lim_{x \to -\infty} \frac{1}{x^n} = 0$
\end{multicols}

\begin{multicols}{3}

\end{itemize}

\subsection*{\underline{\textcolor{darkred}{ 5. Limite à gauche et à droite en un point $x_{0}$ }}}
\subsubsection*{\underline{\textcolor{darkred}{ a. Définition }}}

\textit{Soit $f$ une fonction et $x_0 \in \mathbb{R}$.}

\begin{itemize}
\item \textit{On appelle limite à gauche en $x_0$ de $f$, la limite de $f$ lorsque $x$ tend vers $x_0$, en restant à gauche de $x_0$. On la note :}
\[
\lim_{x \to x_0^-} f(x) \quad \text{\textit{ou}} \quad \lim_{\substack{x \to x_0 \\ x < x_0}} f(x)
\]

\vspace{0.3cm}
\item \textit{On appelle limite à droite en $x_0$ de $f$, la limite de $f$ lorsque $x$ tend vers $x_0$, en restant à droite de $x_0$. On la note :}
\[
\lim_{x \to x_0^+} f(x) \quad \text{\textit{ou}} \quad \lim_{\substack{x \to x_0 \\ x > x_0}} f(x)
\]

\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}

Soit $f$ la fonction définie par : 
$
f(x) =
\begin{cases} 
2x - 1 & \text{\textit{si }} x > 1 \\ 
x^2 + x + 1 & \text{\textit{si }} x \leq 1 
\end{cases}
$

\vspace{0.3cm}

\noindent \textit{Calculons les limites en $1^-$ et en $1^+$ de $f$.} \\
\textit{On a :}

\begin{itemize}
    \item .
    
    $
    \begin{aligned}
    \displaystyle \lim_{x \to 1^-} f(x) &= \lim_{x \to 1^-} x^2 + x + 1 \\
    &= (1)^2 + (1) + 1 \\
    &= 3
    \end{aligned}
    $
    
    $\boxed{\lim_{x \to 1^-} f(x) = 3}$
    
    \vspace{0.3cm}
    \item .
    
     $
    \begin{aligned}
    \displaystyle \lim_{x \to 1^+} f(x) &= \lim_{x \to 1^+} 2x - 1 \\
    &= 2(1) - 1 \\
    &= 1
    \end{aligned}
    $
    
    $ \boxed{\lim_{x \to 1^+} f(x) = 1}$
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ b. Propriété }}}

Pour tout réel $a$, on a :
\[
\lim_{x \to a^+} \frac{1}{x - a} = +\infty \quad ; \quad \lim_{x \to a^-} \frac{1}{x - a} = -\infty
\]

\begin{center}
\begin{tikzpicture}[node style/.style={fill opacity=0,text opacity=1}]
        \tkzTabInit[espcl=2,lgt=3]{$x$/.5,$x - a$/.7}{$-\infty$,$a$ ,$+\infty$}
        \tkzTabLine{ ,-,z,+, }
\end{tikzpicture}
\end{center}

\paragraph{\underline{Explication :}}
Le tableau de signes de l'expression $x - a$ permet de déterminer si le dénominateur tend vers $0$ par valeurs positives ($0^+$) ou par valeurs négatives ($0^-$) :

\begin{itemize}
    \item \textbf{Limite à droite ($x \to a^+$) :} 
    Lorsque $x > a$ (à droite de $a$), le tableau indique que $x - a$ est \textbf{positif} ($+$). Le dénominateur tend donc vers $0$ par valeurs supérieures, soit $0^+$.
    \[
    \text{Par règle des signes : } \lim_{x \to a^+} \frac{1}{x - a} = \frac{1}{0^+} = +\infty
    \]

    \item \textbf{Limite à gauche ($x \to a^-$) :} 
    Lorsque $x < a$ (à gauche de $a$), le tableau indique que $x - a$ est \textbf{négatif} ($-$). Le dénominateur tend donc vers $0$ par valeurs inférieures, soit $0^-$.
    \[
    \text{Par règle des signes : } \lim_{x \to a^-} \frac{1}{x - a} = \frac{1}{0^-} = -\infty
    \]
\end{itemize}
\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}    

\noindent
\begin{multicols}{2}

\noindent \textbf{1. Limites de $\dfrac{1}{x}$ en $0$ :}

\begin{center}
\begin{tikzpicture}[node style/.style={fill opacity=0,text opacity=1}]
    \tkzTabInit[espcl=1.5,lgt=2]{$x$/.5,$x$/.7}{$-\infty$,$0$,$+\infty$}
    \tkzTabLine{ ,-,z,+, }
\end{tikzpicture}
\end{center}

\begin{itemize}
    \item \textbf{À droite ($x \to 0^+$) :} d'après le tableau, pour $x > 0$, le dénominateur est positif ($+$). Donc $x \to 0^+$.
    \[
    \lim_{x \to 0^+} \frac{1}{x} = \frac{1}{0^+} = +\infty
    \]
    \item \textbf{À gauche ($x \to 0^-$) :} d'après le tableau, pour $x < 0$, le dénominateur est négatif ($-$). Donc $x \to 0^-$.
    \[
    \lim_{x \to 0^-} \frac{1}{x} = \frac{1}{0^-} = -\infty
    \]
\end{itemize}

\columnbreak

\noindent \textbf{2. Limites de $\dfrac{1}{x - 2}$ en $2$ :}

\begin{center}
\begin{tikzpicture}[node style/.style={fill opacity=0,text opacity=1}]
    \tkzTabInit[espcl=1.5,lgt=2]{$x$/.5,$x - 2$/.7}{$-\infty$,$2$,$+\infty$}
    \tkzTabLine{ ,-,z,+, }

\end{center}

\begin{itemize}
    \item \textbf{À droite ($x \to 2^+$) :} d'après le tableau, pour $x > 2$, l'expression $x - 2$ est positive ($+$). Le dénominateur tend vers $0^+$.
    \[
    \lim_{x \to 2^+} \frac{1}{x - 2} = \frac{1}{0^+} = +\infty
    \]
    \item \textbf{À gauche ($x \to 2^-$) :} d'après le tableau, pour $x < 2$, l'expression $x - 2$ est négative ($-$). Le dénominateur tend vers $0^-$.
    \[
    \lim_{x \to 2^-} \frac{1}{x - 2} = \frac{1}{0^-} = -\infty
    \]
\end{itemize}

\end{multicols}

\subsection*{\underline{\textcolor{darkred}{ 6. Existence de limite d’une fonction en un point $x_{0}$ }}}
\subsubsection*{\underline{\textcolor{darkred}{ b. Propriété }}}
Soit $f$ une fonction et $x_0 \in \mathbb{R}$. La fonction $f$ admet une limite $l$ en $x_0$ si et seulement si on a :
\[
\lim_{x \to x_0^+} f(x) = \lim_{x \to x_0^-} f(x) = l
\]

\vspace{0.3cm}
\noindent{\Large\bfseries\color{darkred} Méthode :}

Pour déterminer si une fonction $f$ admet ou non une limite en un point $x_0$, on calcule la limite à gauche en $x_0$ et la limite à droite en $x_0$ de $f$, puis on compare les résultats.
\begin{itemize}
\item Si $\displaystyle \lim_{x \to x_0^+} f(x) = \lim_{x \to x_0^-} f(x)$, alors $f$ admet une limite $l$ en $x_0$ vérifiant :
\[
l = \lim_{x \to x_0^+} f(x) = \lim_{x \to x_0^-} f(x).
\]

\vspace{0.2cm}
\item Si $\displaystyle \lim_{x \to x_0^+} f(x) \neq \lim_{x \to x_0^-} f(x)$, alors $f$ n'admet pas de limite en $x_0$.
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}}  

\noindent 1. \textit{Soit $f$ la fonction définie par : } 
$\begin{cases} f(x) = x - 1 & \text{\textit{si }} x > 0 \\ f(x) = x^2 + 1 & \text{\textit{si }} x \leq 0 \end{cases}.$

\vspace{0.1cm}

\noindent \textit{La fonction $f$ admet-elle une limite en $0$ ?} \\
\textit{On a :}

\begin{itemize}
    \item $\displaystyle \lim_{x \to 0^-} f(x) = \lim_{x \to 0^-} x^2 + 1 = (0)^2 + 1 = 1$
    \vspace{0.2cm}
    \item $\displaystyle \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} x - 1 = (0) - 1 = -1$
\end{itemize}

\noindent \textit{Ainsi, $\displaystyle \lim_{x \to 0^+} f(x) \neq \lim_{x \to 0^-} f(x)$, donc $f$ n'admet pas de limite en $0$.}

\vspace{0.6cm}

\noindent 2. \textit{Soit $g$ la fonction définie par : } 
$\begin{cases} g(x) = x - 1 & \text{\textit{si }} x > 1 \\ g(x) = x^2 + x - 2 & \text{\textit{si }} x \leq 1 \end{cases}.$

\vspace{0.1cm}

\noindent \textit{La fonction $g$ admet-elle une limite en $1$ ?} \\
\textit{On a :}

\begin{itemize}
    \item $\displaystyle \lim_{x \to 1^-} g(x) = \lim_{x \to 1^-} x^2 + x - 2 = (1)^2 + (1) - 2 = 0$
    \vspace{0.2cm}
    \item $\displaystyle \lim_{x \to 1^+} g(x) = \lim_{x \to 1^+} x - 1 = (1) - 1 = 0$
\end{itemize}

\noindent \textit{Ainsi, $\displaystyle \lim_{x \to 1^+} g(x) = \lim_{x \to 1^-} g(x) = 0$, d'où $g$ admet une limite en $1$ et on a :}
\[
\lim_{x \to 1} g(x) = 0.
\]

\section*{\underline{\textcolor{darkred}{ II. Limites et opérations sur les fonctions}}}
\begin{itemize}
    \item Les propriétés suivantes, représentées sous forme de tableaux donnent les limites en un point $x_0$. Toutefois les résultats restent vrais en $+\infty$, $-\infty$, et en $x_0$ par valeurs supérieures ou inférieures.

\subsubsection*{\underline{\textcolor{darkred}{ 1. Limite d’une somme de deux fonctions }}}
\subsubsection*{\underline{\textcolor{darkred}{ a. Propriété }}}

        \hline
    \end{tabular}
    \end{center}
    
    \vspace{0.2cm}
    
    \[
    \lim_{x \to 1^+} \frac{x - 3}{x - 1} :
    \begin{cases}
    \displaystyle \lim_{x \to 1^+} (x - 3) = -2 \\
    & \text{Par quotient} \\
    \displaystyle \lim_{x \to 1^+} (x - 1) = 0^+
    \end{cases}
    \]
    
    \[
    \boxed{\lim_{x \to 1^+} \frac{x - 3}{x - 1} = -\infty}
    \]

    \vspace{0.4cm}

    \item \textit{Calculons $\displaystyle \lim_{x \to +\infty} \frac{x + 1}{x - 1}$ :}
    
    \vspace{0.2cm}
    
    \[
    \lim_{x \to +\infty} \frac{x + 1}{x - 1} :
    \begin{cases}
    \displaystyle \lim_{x \to +\infty} (x + 1) = +\infty \\
    & \text{Par quotient (F.I)} \\
    \displaystyle \lim_{x \to +\infty} (x - 1) = +\infty
    \end{cases}
    \]
    
    \vspace{0.2cm}
    \textit{C'est une Forme Indéterminée (F.I) ; on ne peut pas conclure directement.}

    \vspace{0.4cm}

    \item \textit{Calculons $\displaystyle \lim_{x \to 1} \frac{x^2 + 1}{x - 3}$ :}
    
    \vspace{0.2cm}
    
    \[
    \lim_{x \to 1} \frac{x^2 + 1}{x - 3} :
    \begin{cases}
    \displaystyle \lim_{x \to 1} (x^2 + 1) = 2 \\
    & \text{Par quotient} \\
    \displaystyle \lim_{x \to 1} (x - 3) = -2
    \end{cases}
    \]
    
    \[
    \boxed{\lim_{x \to 1} \frac{x^2 + 1}{x - 3} = -1}
    \]
\end{enumerate}

\subsubsection*{\underline{\textcolor{darkred}{ 4. Limites de $\sqrt{u}$ et $|u|$} }}
\subsubsection*{\underline{\textcolor{darkred}{ a. Propriété }}}

Soit $u$ une fonction positive et $x_0$ un réel.

\vspace{0.3cm}

\begin{itemize}
\item Si $\displaystyle \lim_{x \to x_0} u = l$, alors $\displaystyle \lim_{x \to x_0} \sqrt{u} = \sqrt{l}$
\item Si $\displaystyle \lim_{x \to x_0} u = +\infty$, alors $\displaystyle \lim_{x \to x_0} \sqrt{u} = +\infty$
\end{itemize}

\subsubsection*{\underline{\textcolor{darkred}{ Exemple }}} 

\begin{enumerate}
    \item \textit{Soit $g$ la fonction définie par : $g(x) = \sqrt{x + 3}$. Calculons $\displaystyle \lim_{x \to 1} g(x)$.}
    
    \vspace{0.2cm}
    
    \textit{On a : $\displaystyle \lim_{x \to 1} x + 3 = 4$ donc $\displaystyle \lim_{x \to 1} g(x) = \sqrt{4} = 2$.}
    
    \[
    \boxed{\lim_{x \to 1} g(x) = 2}
    \]
\end{enumerate}
