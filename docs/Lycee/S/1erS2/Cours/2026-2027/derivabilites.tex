\section*{I. Fonction dérivable en un point}

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=1. Définition du nombre dérivé,
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Soit $f$ une fonction numérique et $a$ un nombre réel tel que $f$
est définie sur un intervalle $I$ contenant $a$. On dit que $f$
est dérivable en $a$ si :

\[
\lim_{x\to a}\frac{f(x)-f(a)}{x-a}=\ell\in\mathbb{R}
\]

$\ell$ est appelé nombre dérivé de $f$ en $a$ que l'on note
$f'(a)=\ell$.

\end{tcolorbox}

\textbf{Exemples :} Les fonctions suivantes sont-elles dérivables
aux points indiqués ?

\[
f(x)=x^2 \quad \text{en } a=-1
\qquad\qquad
g(x)=2x^3-1 \quad \text{en } a=2
\]

\textbf{Contre exemple :} $f(x)=|x-1|$ et $f(1)=0$.
$f$ est-elle dérivable en $1$ ?

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{2. Théorème},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Si une fonction $f$ est dérivable en un réel $a$, alors $f$ est
continue en $a$. C'est-à-dire que la dérivabilité entraîne
la continuité.

\textbf{NB :} La réciproque n'est pas toujours vraie.

\end{tcolorbox}

\textbf{Exemple :}

\begin{enumerate}

    \item Soit
    \[
    g(x)=\frac{f(x)-f(1)}{x-1}
    \]
    avec $f(x)=|x-1|$. $g$ est-elle continue en $1$ ?

    \item Soit $f(x)=|x|$. $f$ est-elle dérivable en $0$ ?

\end{enumerate}

% -------------------------------------------------
% 3. Interprétation graphique
% -------------------------------------------------

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{3. Interprétation graphique},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Soit $f$ une fonction numérique dérivable en un réel $a$.
La courbe représentative de la fonction $f$ admet une tangente
d'équation :

\[
(T):\quad y=f'(a)(x-a)+f(a)
\quad \text{au point } A(a;f(a))
\]

Le coefficient directeur de $(T)$ est $f'(a)$.

\end{tcolorbox}

\textbf{Exemples :}

\begin{enumerate}

    \item $f(x)=x^2-3x-1,\qquad x_0=-2$

    \item $f(x)=\dfrac{2-3x}{x-2},\qquad x_0=0$

    \item $f(x)=\sqrt{2x-3},\qquad x_0=2$

\end{enumerate}

% -------------------------------------------------
% 4. Dérivée à gauche et dérivée à droite
% -------------------------------------------------

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{4. Dérivée à gauche et dérivée à droite},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Soit $f$ et $a$ un nombre réel :

\begin{itemize}

    \item Si
    \(
    \lim\limits_{x\to a^-}\frac{f(x)-f(a)}{x-a}
    =\ell\in\mathbb{R}
    \)
    alors on dit que $f$ est dérivable à gauche de $a$ et le réel
    $\ell$ est appelé nombre dérivé de $f$ à gauche de $a$ que l'on
    note $f'_g(a)=\ell$.

    \item Si
    \(
    \lim\limits_{x\to a^+}\frac{f(x)-f(a)}{x-a}
    =\ell'\in\mathbb{R}
    \)
    alors on dit que $f$ est dérivable à droite de $a$ et que
    $\ell'$ est le nombre dérivé de $f$ à droite de $a$ que l'on
    note $f'_d(a)=\ell'$.

\end{itemize}

\end{tcolorbox}

% -------------------------------------------------
% Théorème
% -------------------------------------------------

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{Théorème :},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

\begin{itemize}

    \item Si $f'_g(a)=f'_d(a)$ alors $f$ est dérivable en $a$.

    \item Si $f'_g(a)\neq f'_d(a)$ alors $f$ n'est pas dérivable en $a$.

    \item Si
    \(
    \lim\limits_{x\to a}\frac{f(x)-f(a)}{x-a}=\infty
    \)
    alors $f$ n'est pas dérivable en $a$.

\end{itemize}

\end{tcolorbox}

\textbf{Exemple :}

Soit
\[
f(x)=
\begin{cases}
x^2-1 & \text{si } x<1,\\[1mm]
\dfrac{x-1}{x+1} & \text{si } x\geq 1.
\end{cases}
\quad
\text{. Étudier la dérivabilité de } f \text{ en } 1.
\]

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{Interprétation graphique},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

\begin{itemize}

    \item Si $f$ est dérivable à gauche de $a$ alors $\mathcal{C}_f$
    admet une demi-tangente à gauche de $a$ d'équation
    $(T_g):\ y=f'_g(a)(x-a)+f(a)$ au point $A(a;f(a))$.

    \item Si $f$ est dérivable à droite de $a$ alors $\mathcal{C}_f$
    admet une demi-tangente à droite de $a$ d'équation
    $(T_d):\ y=f'_d(a)(x-a)+f(a)$ au point $A(a;f(a))$.

\end{itemize}

\end{tcolorbox}

\textbf{Exemple :}

Soit
\[
f(x)=
\begin{cases}
x^2-1 & \text{si } x<1,\\[1mm]
\dfrac{x-1}{x+1} & \text{si } x\geq 1.
\end{cases}
\quad \text{. Étudier la dérivabilité de } f \text{ en } 1
\text{ puis interpréter graphiquement les résultats.}
\]

\textbf{Solution :} Puisque $f'_g(1)\neq f'_d(1)$, alors $\mathcal{C}_f$
admet deux demi-tangentes d'équations respectives :

\[
\begin{aligned}
(T_g):\quad y
&= f'_g(1)(x-1)+f(1)
&&\Longrightarrow&
(T_g):\quad y &= 2x-2
\\[2mm]
(T_d):\quad y
&= f'_d(1)(x-1)+f(1)
&&\Longrightarrow&
(T_d):\quad y &= \frac{1}{2}x-\frac{1}{2}
\end{aligned}
\]

\textbf{Tableaux de valeurs pour tracer les demi-tangentes :}

\[
\begin{array}{c@{\qquad\qquad}c}
(T_g):\ y=2x-2
&
(T_d):\ y=\dfrac{1}{2}x-\dfrac{1}{2}
\\[2mm]
\begin{array}{|c|c|c|}
\hline
x & 0 & 1\\
\hline
y & -2 & 0\\
\hline
\end{array}
&
\begin{array}{|c|c|c|}
\hline
x & 0 & 1\\
\hline
y & -\dfrac{1}{2} & 0\\
\hline
\end{array}
\end{array}
\]

\vspace{0.3cm}

\textbf{Tableaux de valeurs pour les courbes :}

\[
\begin{array}{c@{\qquad\qquad}c}
\begin{array}{c}
x^2-1
\\[1mm]
\begin{array}{|c|c|c|c|c|}
\hline
x & -2 & -1 & 0 & 1\\
\hline
f(x) & 3 & 0 & -1 & 0\\
\hline
\end{array}
\end{array}
&
\begin{array}{c}
\dfrac{x-1}{x+1}
\\[1mm]
\begin{array}{|c|c|c|c|c|}
\hline
x & 0 & 1 & 2 & 3\\
\hline
f(x) & -1 & 0 & \dfrac{1}{3} & \dfrac{1}{2}\\
\hline
\end{array}
\end{array}
\end{array}
\]

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{Remarque :},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

\begin{align*}
\text{---}\quad
&\lim_{x\to a^\pm}\frac{f(x)-f(a)}{x-a}
=\pm\infty
\quad \text{alors } \mathcal{C}_f \text{ admet une demi-tangente verticale}\\
&\hspace{1.2cm}\text{dirigée vers le haut.}
\\[1mm]
\text{---}\quad
&\lim_{x\to a^\pm}\frac{f(x)-f(a)}{x-a}
=+\infty
\quad \text{alors } \mathcal{C}_f \text{ admet une demi-tangente verticale}\\
&\hspace{1.2cm}\text{dirigée vers le bas.}
\end{align*}

\end{tcolorbox}

\section*{II. Fonction dérivée}

% ---------------------------------------------------------
% 1. Dérivée sur un intervalle
% ---------------------------------------------------------

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{1. Dérivé sur un intervalle},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Soit $f$ une fonction numérique et $I$ un intervalle
$I=]a,b[$ tel que $f$ y est définie.

On dit que $f$ est dérivable sur $I$ si $f$ est dérivable en
chaque point appartenant à $I$.

Si $f$ est définie sur un intervalle $J=[a,b]$ alors on dit que
$f$ est dérivable sur $J$ si $f$ est dérivable en chaque point
appartenant à $J$ et de plus $f$ dérivable à gauche de $b$ et à
droite de $a$.

Si $f$ est dérivable sur un intervalle $I$ alors la fonction
définie sur $I$ par $f'$ qui à tout réel $x\in I$ associe $f'(x)$
le nombre dérivé de $f$ en $x$. On dit que $f'$ est la dérivée
(dérivée) de $f$ en $x$.

\end{tcolorbox}

% ---------------------------------------------------------
% 2. Dérivée des fonctions usuelles
% ---------------------------------------------------------

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{2. Dérivée des fonctions usuelles},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Le tableau ci-dessous nous donne la dérivée de fonctions usuelles.

\begin{center}

\begin{tabular}{|c|c|c|}

\\[1.2mm]

\\[1.2mm]
\hline

$f(x)=ax+b$

&
$\mathbb{R}$
\\[1.2mm]
\hline

$f(x)=x^n\quad (n\in\mathbb{N})$
&
$f'(x)=nx^{n-1}$
&
$\mathbb{R}$
\\[1.2mm]
\hline

$f(x)=\dfrac{1}{x}$
&
$f'(x)=-\dfrac{1}{x^2}$
&
$\mathbb{R}^*$
\\[1.2mm]
\hline

$f(x)=\sqrt{x}$

\\[1.2mm]
\hline

$f(x)=\cos x$

$f'(x)=-\sin x$
&
$\mathbb{R}$
\\[1.2mm]

$\mathbb{R}$
\\[1.2mm]
\hline

\end{tabular}

\end{center}

\end{tcolorbox}

\textbf{Exemples :}

\begin{itemize}

    \item $f(x)=\sqrt{2}
    \quad\Longrightarrow\quad
    f'(x)=0$

    \item $f(x)=2x+3
    \quad\Longrightarrow\quad
    f'(x)=2$

    \item $f(x)=x^3
    \quad\Longrightarrow\quad
    f'(x)=3x^2$

    \item $f(x)=\dfrac{3}{4}x^5+x^2
    \quad\Longrightarrow\quad
    f'(x)=\dfrac{3}{4}\times5x^4+2x
    =\dfrac{15}{4}x^4+2x$

    \item $f(x)=-5x
    \quad\Longrightarrow\quad
    f'(x)=-5$

    \item $f(x)=-5x^4+9
    \quad\Longrightarrow\quad
    f'(x)=-20x^3$

\end{itemize}


\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{Remarque :},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Les fonctions numériques sont dérivables sur l'ouvert de leur
ensemble de définition.

\end{tcolorbox}

\textbf{Exemple :}

\[
\begin{aligned}
f(x)&=\sqrt{x}
&\quad\Longrightarrow\quad&
f\text{ existe ssi }x\geq0
&\quad\Longrightarrow\quad&
D_f=[0,+\infty[
\\[2mm]
f(x)&=\sqrt{x}
&\quad\Longrightarrow\quad&
f'(x)=\dfrac{1}{2\sqrt{x}}
&\quad&
f'\text{ existe ssi }x>0
&\quad\Longrightarrow\quad&
\boxed{D_{f'}=]0,+\infty[}
\end{aligned}
\]

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{3 - Opérations sur les dérivées},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Soit $u$ et $v$ deux fonctions numériques dérivables sur un intervalle
$I$. Pour obtenir la dérivée de la somme, du produit, du quotient et
de la composée, on se réfère au tableau suivant :

\vspace{3mm}

\begin{center}
\renewcommand{\arraystretch}{1.3}

\begin{tabular}{|c|c|c|c|c|c|}
\hline
& 1 & 2 & 3 & 4 & 5
\\
\hline

$f$ définie par
&
$ku\;(k\in\mathbb{R}^*)$
&
$u+v$
&
$u\cdot v$
&
$\dfrac{u}{v}$
&
$\dfrac{1}{v}$
\\
\hline

dérivée de $f$
&
$ku'$
&
$u'+v'$
&
$u'v+v'u$
&
$\dfrac{u'v- v'u}{v^2}$
&
$-\dfrac{v'}{v^2}$
\\
\hline

\end{tabular}

\vspace{6mm}

\begin{tabular}{|c|c|c|c|c|c|}
\hline
& 6 & 7 & 8 & 9 & 10
\\
\hline

&
$\sqrt{u}$
&
$u^n$
&
$u\circ v$
&
$\cos(u)$
&
$\sin(u)$
\\
\hline

&
$\dfrac{u'}{2\sqrt{u}}$
&
$nu'u^{n-1}$
&
$v'\times u'\circ v$
&
$-u'\sin(u)$
&
$u'\cos(u)$
\\
\hline

\end{tabular}

\end{center}

\end{tcolorbox}

\textbf{Exemples :}

\begin{enumerate}

    \item $f(x)=2(x^2-3x+1)
    \quad\Longrightarrow\quad
    f'(x)=2(2x-3)$

    \item $f(x)=(4x^2+5x)+(7x-9)
    \quad\Longrightarrow\quad
    f'(x)=(8x+5)+7=8x+12$

    \item $f(x)=(3x-5)(x^2-1)
    \quad\Longrightarrow\quad
    \begin{aligned}[t]
    f'(x)
    &=3(x^2-1)+2x(3x-5)\\
    &=3x^2-3+6x^2-10x\\
    &=9x^2-10x-3
    \end{aligned}$

    \item $f(x)=\dfrac{2x^2-1}{x+1}
    \quad\Longrightarrow\quad
    f'(x)=
    \dfrac{4x(x+1)-1(2x^2-1)}{(x+1)^2}
    =\dfrac{2x^2+4x+1}{(x+1)^2}$

    \item $f(x)=\dfrac{1}{2x-3}
    \quad\Longrightarrow\quad
    f'(x)=-\dfrac{2}{(2x-3)^2}$

    \item $f(x)=\sqrt{3x+1}
    \quad\Longrightarrow\quad
    f'(x)=\dfrac{3}{2\sqrt{3x+1}}$

    \item $f(x)=(x^2-1)^3
    \quad\Longrightarrow\quad
    f'(x)=3(2x)(x^2-1)^2
    =6x(x^2-1)^2$

    \item
    \[
    f(x)=\sqrt{(x-1)^3}
    \quad\Longrightarrow\quad
    \begin{cases}
        u=\sqrt{x} \quad\Longrightarrow\quad
        u'=\dfrac{1}{2\sqrt{x}},\\[2mm]
        v=(x-1)^3 \quad\Longrightarrow\quad
        v'=3(x-1)^2.
    \end{cases}
    \]

    \[
    f'(x)=3(x-1)^2\times
    \dfrac{1}{2\sqrt{(x-1)^3}}
    \]

    \item
    \[
    \begin{aligned}
    f(x)&=\cos x
    \quad\Longrightarrow\quad
    f'(x)=-1\sin x=-(x)'\sin x
    \\[2mm]
    f(2x-3)&=\cos(2x-3)
    \quad\Longrightarrow\quad
    f'(2x-3)=-2\sin(2x-3)
    \end{aligned}
    \]

    \item
    \[
    \begin{aligned}
    f(x)&=\sin x
    \quad\Longrightarrow\quad
    f'(x)=1\cos x=(x)'\cos x
    \\[2mm]
    f(3x^2-5x+4)&=\sin(3x^2-5x+4)
    \quad\Longrightarrow\quad
    f'(v)=(6x-5)\cos(3x^2-5x+4)
    \end{aligned}
    \]

\end{enumerate}

\section*{III - Application de la dérivation}

% ---------------------------------------------------------
% 1 - Sens de variation
% ---------------------------------------------------------

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{1 - Sens de variation},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Soit $f$ une fonction numérique dérivable sur un intervalle $I$.

\vspace{2mm}

\textbf{$\star$ Théorème :}

\begin{itemize}

    \item[$\text{---}$] Si $f'(x)\geq 0$ alors $f$ est croissante sur $I$.

    \item[$\text{---}$] Si $f'(x)\leq 0$ alors $f$ est décroissante sur $I$.

    \item[$\text{---}$] Si $f'(x)>0$ alors $f$ est strictement croissante sur $I$.

    \item[$\text{---}$] Si $f'(x)<0$ alors $f$ est strictement décroissante sur $I$.

    \item[$\text{---}$] Si $f$ est dérivable sur $I$ et
    $\exists x_0\in I/f'(x_0)=0$ en changement de signe alors on dit
    que $f$ admet un extremum.

\end{itemize}

\end{tcolorbox}

% =========================================================
% Remarque
% =========================================================

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{Remarque :},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

\[
\begin{aligned}
\text{---}\quad &\text{Si on passe de $+$ en $-$, alors on obtient un maximum relatif.}
\\[2mm]
\text{---}\quad &\text{Si on passe de $-$ en $+$, alors on obtient un minimum relatif.}
\end{aligned}
\]

\textbf{NB :} Étudier la monotonie d'une fonction numérique c'est
déterminer son sens de variation. La monotonie s'étudie uniquement
sur un intervalle mais pas sur une réunion d'intervalles.

\end{tcolorbox}


% =========================================================
% Exemples
% =========================================================

\textbf{Exemples :}

Étudier le sens de variation des fonctions suivantes sur leurs domaines :

\begin{enumerate}
    \item $f(x)=2x^3-6x+1$
    
    \item $g(x)=\sqrt{x^2-x+1}$
    
    \item $h(x)=\dfrac{3x-1}{2x+3}$
\end{enumerate}


% =========================================================
% 2 - Tableau de variation
% =========================================================
\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{2 - Tableau de variation},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Un tableau de variation c'est un tableau ayant trois lignes dont chaque ligne a des caractéristiques bien définies :

\[
\begin{aligned}
\text{---}\quad &\text{La 1\textsuperscript{re} ligne est constituée par :
Le domaine de définition et les valeurs qui annulent}\\
&\hspace{1.1cm}\text{la dérivée.}
\\[2mm]
\text{---}\quad &\text{La 2\textsuperscript{e} ligne est constituée du signe de la dérivée.}
\\[2mm]
\text{---}\quad &\text{La 3\textsuperscript{e} ligne est constituée des :
limite aux bornes, les extremums et les variations.}
\end{aligned}
\]
\end{tcolorbox}

\textbf{Exemple :}

Soit
\[
f(x)=\dfrac{2x^2-1}{x+1}.
\]
Dresser le tableau de variation de $f$.


% =========================================================
% 3 - Dérivée et bijection
% =========================================================

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{3 - Dérivée et bijection},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Soit $f$ une fonction numérique définie sur un intervalle $I$.
Si $f$ est continue et dérivable sur l'intervalle $I$ et que $f$
y est monotone alors $f$ réalise une bijection ou admet une
bijection réciproque $f^{-1}$ définie de $J=f(I)$ vers $I$.
C'est-à-dire :

\[
\begin{aligned}
f &: I \longrightarrow J=f(I)
\\
f^{-1} &: J=f(I) \longrightarrow I
\end{aligned}
\]

\end{tcolorbox}


% =========================================================
% Remarque
% =========================================================

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{Remarque :},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

$f$ et $f^{-1}$ ont même sens de monotonie.

\end{tcolorbox}


% =========================================================
% Théorème
% =========================================================

\begin{tcolorbox}[
    colback=green!5,
    colframe=green!60!black,
    colbacktitle=green!60!black,
    title=\textbf{Théorème :},
    fonttitle=\bfseries,
    boxrule=0.8pt
]

Si $f$ admet une bijection réciproque $f^{-1}$ sur $I$ et que
$f$ s'annule sur $I$ alors l'équation $f(x)=0$ admet une unique
solution $\alpha$ tel que $0\in f(I)=J$.

\end{tcolorbox}
