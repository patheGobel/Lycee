\section*{\color{myred}I. Notion de primitive d'une fonction}

\subsection*{\color{myred}1. Définition}

\begin{tcolorbox}[colback=blue!5!white,colframe=myblue,title=\textbf{Définition}]
Soit $f$ une fonction définie sur un intervalle $I$.\\
On appelle \textbf{primitive} de $f$ sur $I$, toute fonction $F$ dérivable sur $I$ telle que pour tout $x \in I$ :
\[
F'(x) = f(x)
\]
\end{tcolorbox}

\paragraph{Exemple}
Soit $f(x) = 2x$ définie sur $\mathbb{R}$.\\
Si on pose $F(x) = x^2$, on remarque que $F'(x) = 2x = f(x)$.\\
La fonction $F(x) = x^2$ est donc une primitive de $f(x) = 2x$ sur $\mathbb{R}$.

\subsection*{\color{myred}2. Ensemble des primitives et condition initiale}

\paragraph{Activité préparatoire :}
Considérons les trois fonctions polynômes suivantes définies sur $\mathbb{R}$ :
\[
F_1(x) = x^2, \quad F_2(x) = x^2 + 3 \quad \text{et} \quad F_3(x) = x^2 - 5
\]
Calculons la dérivée de chacune de ces fonctions :
\begin{itemize}
    \item $F_1'(x) = (x^2)' = 2x$
    \item $F_2'(x) = (x^2 + 3)' = 2x + 0 = 2x$
    \item $F_3'(x) = (x^2 - 5)' = 2x - 0 = 2x$
\end{itemize}

\textbf{Constat :} Bien que les constantes soient différentes ($0$, $3$ et $-5$), la dérivée de ces trois polynômes donne exactement le même polynôme $f(x) = 2x$.\\
Ainsi, $F_1$, $F_2$ et $F_3$ sont toutes des primitives de la même fonction $f$. De manière générale, ajouter une constante réelle $k$ ne modifie pas la dérivée car $(k)' = 0$.

\vspace{0.2cm}

\begin{tcolorbox}[colback=red!5!white,colframe=myred,title=\textbf{Propriétés}]
\begin{enumerate}[label=\arabic*.]
    \item Si $F$ est une primitive de $f$ sur $I$, alors toutes les autres primitives de $f$ sont de la forme :
    \[
    x \mapsto F(x) + k \quad \text{où } k \in \mathbb{R} \text{ est une constante réelle.}
    \]
    \item Il existe une \textbf{unique primitive} $F$ de $f$ qui prend une valeur donnée $y_0$ en un point $x_0$ ($F(x_0) = y_0$).
\end{enumerate}
\end{tcolorbox}

\paragraph{Exemple :}
Pour la fonction $f(x) = 2x$, toutes les primitives sont de la forme $F(x) = x^2 + k$.\\
Trouvons la primitive $F$ telle que $F(1) = 5$ :
\[
F(1) = 5 \iff 1^2 + k = 5 \iff 1 + k = 5 \iff k = 4
\]
La primitive unique cherchée est donc $F(x) = x^2 + 4$.

\section*{\color{myred}II. Primitives des fonctions usuelles}

\begin{center}
\renewcommand{\arraystretch}{1.9}
\begin{tabular}{|c|c|c|}
\hline
\textbf{Fonction } $f(x)$ & \textbf{Une primitive } $F(x)$ & \textbf{Intervalle de validité} \\
\hline
$a \quad (a \in \mathbb{R})$ & $ax$ & $\mathbb{R}$ \\
\hline
$x$ & $\dfrac{1}{2}x^2$ & $\mathbb{R}$ \\
\hline
$x^n \quad (n \in \mathbb{N}^*)$ & $\dfrac{1}{n+1}x^{n+1}$ & $\mathbb{R}$ \\
\hline
$\dfrac{1}{x^2}$ & $-\dfrac{1}{x}$ & $]-\infty \,;\, 0[$ ou $]0 \,;\, +\infty[$ \\
\hline
$\dfrac{1}{\sqrt{x}}$ & $2\sqrt{x}$ & $]0 \,;\, +\infty[$ \\
\hline
\end{tabular}
\end{center}

\paragraph{Règles d'opération :}
Si $F$ est une primitive de $f$ et $G$ est une primitive de $g$, alors :
\begin{itemize}
    \item Une primitive de $f + g$ est $F + G$.
    \item Une primitive de $k \times f$ est $k \times F$ (avec $k \in \mathbb{R}$).
\end{itemize}

\subsection*{\color{myred}4. Exercice d'application récapitulatif}

\begin{tcolorbox}[colback=yellow!5!white,colframe=myred,title=\textbf{Exercice d'entraînement}]
Dans chacun des cas suivants, déterminer une primitive $F$ de la fonction $f$ sur l'intervalle $I$ indiqué :

\begin{enumerate}
    \item $f(x) = 4x^3 - 6x^2 + 2x - 7$ sur $I = \mathbb{R}$.
    \item $f(x) = 3 + \dfrac{2}{x^2} - \dfrac{1}{\sqrt{x}}$ sur $I = ]0 \,;\, +\infty[$.
    \item $f(x) = \dfrac{6x - 2}{3x^2 - 2x + 5}$ sur $I = \mathbb{R}$.
    \item $f(x) = \dfrac{1}{\sqrt{2x + 1}}$ sur $I = ]-\frac{1}{2} \,;\, +\infty[$.
    \item Déterminer la primitive $G$ de la fonction $g(x) = 3x^2 - 2x + 1$ sur $\mathbb{R}$ telle que $G(1) = 5$.
\end{enumerate}
\end{tcolorbox}

\vspace{0.2cm}

\begin{tcolorbox}[colback=gray!5!white,colframe=myblue,title=\textbf{Correction détaillée}]

\paragraph{1. Fonction polynôme :}
Pour $f(x) = 4x^3 - 6x^2 + 2x - 7$ sur $\mathbb{R}$ :
\[
F_1(x) = 4 \times \frac{x^4}{4} - 6 \times \frac{x^3}{3} + 2 \times \frac{x^2}{2} - 7x = \mathbf{x^4 - 2x^3 + x^2 - 7x + C}
\]

\paragraph{2. Combinaison de formes usuelles :}
Pour $f(x) = 3 + \dfrac{2}{x^2} - \dfrac{1}{\sqrt{x}}$ sur $]0 \,;\, +\infty[$ :
\begin{itemize}
    \item La primitive de $3$ est $3x$.
    \item La primitive de $\dfrac{2}{x^2} = 2 \times \dfrac{1}{x^2}$ est $2 \times \left(-\dfrac{1}{x}\right) = -\dfrac{2}{x}$.
    \item La primitive de $-\dfrac{1}{\sqrt{x}}$ est $-2\sqrt{x}$.
\end{itemize}
D'où : $\mathbf{F_2(x) = 3x - \frac{2}{x} - 2\sqrt{x} + C}$.

\paragraph{3. Forme $\dfrac{u'(x)}{u(x)}$ :}
Pour $f(x) = \dfrac{6x - 2}{3x^2 - 2x + 5}$ sur $\mathbb{R}$ :
\begin{itemize}
    \item Posons $u(x) = 3x^2 - 2x + 5$.
    \item Sa dérivée est $u'(x) = 6x - 2$.
    \item La fonction $f$ est exactement de la forme $\dfrac{u'}{u}$.
\end{itemize}
D'où : $\mathbf{F_3(x) = \ln(3x^2 - 2x + 5) + C}$.

\paragraph{4. Forme $\dfrac{u'(x)}{\sqrt{u(x)}}$ :}
Pour $f(x) = \dfrac{1}{\sqrt{2x + 1}}$ sur $]-\frac{1}{2} \,;\, +\infty[$ :
\begin{itemize}
    \item Posons $u(x) = 2x + 1$, d'où $u'(x) = 2$.
    \item Réécrivons $f(x)$ pour faire apparaître $u'(x)$ au numérateur :
    \[
    f(x) = \frac{1}{2} \times \frac{2}{\sqrt{2x + 1}} = \frac{1}{2} \times \frac{u'(x)}{\sqrt{u(x)}}
    \]
    \item Comme la primitive de $\dfrac{u'}{\sqrt{u}}$ est $2\sqrt{u}$ :
    \[
    F_4(x) = \frac{1}{2} \times \left(2\sqrt{2x + 1}\right) = \mathbf{\sqrt{2x + 1} + C}
    \]
\end{itemize}

\paragraph{5. Recherche d'une primitive particulière (Condition initiale) :}
Pour $g(x) = 3x^2 - 2x + 1$ :
\begin{enumerate}
    \item Primitive générale : $G(x) = 3 \times \dfrac{x^3}{3} - 2 \times \dfrac{x^2}{2} + x + C = x^3 - x^2 + x + C$.
    \item Détermination de $C$ avec $G(1) = 5$ :
    \[
    G(1) = 5 \iff 1^3 - 1^2 + 1 + C = 5 \iff 1 + C = 5 \implies C = 4
    \]
    \item Conclusion : La primitive unique est $\mathbf{G(x) = x^3 - x^2 + x + 4}$.
\end{enumerate}

\end{tcolorbox}
