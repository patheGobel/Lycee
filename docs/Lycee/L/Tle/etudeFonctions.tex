\section*{\color{myred}I. Parité et éléments de symétrie}

Soit $f$ une fonction numérique définie sur un ensemble $D_f$ et $\mathcal{C}_f$ sa courbe représentative dans un repère orthogonal.

\subsection*{\color{myred}1. Parité d'une fonction}

\begin{tcolorbox}[colback=blue!5!white,colframe=myblue,title=\textbf{Définition 1 : Fonction paire}]
On dit qu'une fonction $f$ est \textbf{paire} si :
\begin{enumerate}
    \item Pour tout $x \in D_f$, $-x \in D_f$ ($D_f$ est centré en $0$).
    \item Pour tout $x \in D_f$, $f(-x) = f(x)$.
\end{enumerate}
\textbf{Interprétation géométrique :} L'axe des ordonnées $(Oy)$ est un \textbf{axe de symétrie} pour la courbe $\mathcal{C}_f$.
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5!white,colframe=myred,title=\textbf{Définition 2 : Fonction impaire}]
On dit qu'une fonction $f$ est \textbf{impaire} si :
\begin{enumerate}
    \item Pour tout $x \in D_f$, $-x \in D_f$.
    \item Pour tout $x \in D_f$, $f(-x) = -f(x)$.
\end{enumerate}
\textbf{Interprétation géométrique :} L'origine du repère $O(0,0)$ est un \textbf{centre de symétrie} pour la courbe $\mathcal{C}_f$.
\end{tcolorbox}

\paragraph{Exemple 1 : Étude de la parité}
\begin{itemize}
    \item \textbf{Cas d'une fonction paire :} Soit $f(x) = x^4 - 2x^2 + 1$ définie sur $D_f = \mathbb{R}$.
    \begin{enumerate}
        \item $D_f = \mathbb{R}$ est symétrique par rapport à $0$.
        \item Pour tout $x \in \mathbb{R}$ :
        \[
        f(-x) = (-x)^4 - 2(-x)^2 + 1 = x^4 - 2x^2 + 1 = f(x)
        \]
        Donc $f$ est **paire**, et $(Oy)$ est axe de symétrie de $\mathcal{C}_f$.
    \end{enumerate}

    \item \textbf{Cas d'une fonction impaire :} Soit $g(x) = \dfrac{2x}{x^2 + 1}$ définie sur $D_g = \mathbb{R}$.
    \begin{enumerate}
        \item $D_g = \mathbb{R}$ est symétrique par rapport à $0$.
        \item Pour tout $x \in \mathbb{R}$ :
        \[
        g(-x) = \frac{2(-x)}{(-x)^2 + 1} = \frac{-2x}{x^2 + 1} = -g(x)
        \]
        Donc $g$ est **impaire**, et l'origine $O(0,0)$ est centre de symétrie de $\mathcal{C}_g$.
    \end{enumerate}
\end{itemize}

\subsection*{\color{myred}2. Éléments de symétrie}

\begin{tcolorbox}[colback=gray!5!white,colframe=black,title=\textbf{Propriété 1 : Axe de symétrie}]
La droite $(\mathcal{D}) : x = a$ est un \textbf{axe de symétrie} de la courbe $\mathcal{C}_f$ si et seulement si pour tout $x \in D_f$ :
\[
(2a - x) \in D_f \quad \text{et} \quad f(2a - x) = f(x)
\]
\end{tcolorbox}

\paragraph{Exemple 2 : Démontrer qu'une droite est axe de symétrie}
Soit $f(x) = x^2 - 4x + 5$ définie sur $D_f = \mathbb{R}$. Montrons que la droite $(\mathcal{D}) : x = 2$ est axe de symétrie de $\mathcal{C}_f$.

\paragraph{Résolution :}
Ici $a = 2$. Pour tout $x \in \mathbb{R}$, $(2a - x) = (4 - x) \in \mathbb{R}$.
Calculons $f(4 - x)$ :
\[
f(4 - x) = (4 - x)^2 - 4(4 - x) + 5
\]
\[
f(4 - x) = (16 - 8x + x^2) - 16 + 4x + 5
\]
\[
f(4 - x) = x^2 - 4x + 5 = f(x)
\]
Puisque $f(4 - x) = f(x)$, la droite d'équation $x = 2$ est bien un **axe de symétrie** pour $\mathcal{C}_f$.

\vspace{0.3cm}

\begin{tcolorbox}[colback=gray!5!white,colframe=black,title=\textbf{Propriété 2 : Centre de symétrie}]
Le point $I(a, b)$ est un \textbf{centre de symétrie} de la courbe $\mathcal{C}_f$ si et seulement si pour tout $x \in D_f$ :
\[
(2a - x) \in D_f \quad \text{et} \quad f(2a - x) + f(x) = 2b
\]
\end{tcolorbox}

\paragraph{Exemple 3 : Démontrer qu'un point est centre de symétrie}
Soit $g(x) = \dfrac{2x + 1}{x - 1}$ définie sur $D_g = \mathbb{R} \setminus \{1\}$. Montrons que $I(1, 2)$ est centre de symétrie de $\mathcal{C}_g$.

\paragraph{Résolution :}
Ici $a = 1$ et $b = 2$.
\begin{enumerate}
    \item Si $x \in D_g \iff x \neq 1 \iff -x \neq -1 \iff 2 - x \neq 1$. Donc $(2a - x) \in D_g$.
    \item Calculons $g(2a - x) + g(x) = g(2 - x) + g(x)$ :
    \[
    g(2 - x) = \frac{2(2 - x) + 1}{(2 - x) - 1} = \frac{4 - 2x + 1}{1 - x} = \frac{5 - 2x}{1 - x} = \frac{2x - 5}{x - 1}
    \]
    Additionnons les deux expressions :
    \[
    g(2 - x) + g(x) = \frac{2x - 5}{x - 1} + \frac{2x + 1}{x - 1} = \frac{(2x - 5) + (2x + 1)}{x - 1} = \frac{4x - 4}{x - 1} = \frac{4(x - 1)}{x - 1} = 4
    \]
    Comme $2b = 2(2) = 4$, on a bien $g(2 - x) + g(x) = 2b$.
\end{enumerate}
Le point $I(1, 2)$ est donc bien le **centre de symétrie** de la courbe $\mathcal{C}_g$.

\section*{\color{myred}II. Branches infinies}

\subsection*{\color{myred}1. Asymptotes}

\subsubsection*{\color{myred}a. Asymptote verticale :}

\begin{center}
    \begin{tcolorbox}[colback=blue!5!white,colframe=myblue,title=\textbf{Définition : Asymptote verticale}]
Si la limite d'une fonction $f$ en un réel $a$ est infinie, c'est-à-dire :
\[
\lim_{x \to a^-} f(x) = \pm\infty \quad \text{ou} \quad \lim_{x \to a^+} f(x) = \pm\infty
\]
alors la droite d'équation $\mathbf{x = a}$ est une \textbf{asymptote verticale} à la courbe $\mathcal{C}_f$.
\end{tcolorbox}
\end{center}

\paragraph{Exemple 1 :} Soit $f(x) = \dfrac{3x + 1}{x - 2}$ définie sur $\mathbb{R} \setminus \{2\}$.
\[
\lim_{x \to 2^-} f(x) = \frac{7}{0^-} = -\infty \quad \text{et} \quad \lim_{x \to 2^+} f(x) = \frac{7}{0^+} = +\infty
\]
\textbf{Conclusion :} La droite d'équation $x = 2$ est une asymptote verticale à la courbe $\mathcal{C}_f$.

\subsubsection*{\color{myred}b. Asymptote horizontale :}

\begin{center}
    \begin{tcolorbox}[colback=blue!5!white,colframe=myblue,title=\textbf{Définition : Asymptote horizontale}]
Si la limite d'une fonction $f$ en $+\infty$ ou en $-\infty$ est un nombre réel $L$, c'est-à-dire :
\[
\lim_{x \to \pm\infty} f(x) = L \quad (L \in \mathbb{R})
\]
alors la droite d'équation $\mathbf{y = L}$ est une \textbf{asymptote horizontale} à la courbe $\mathcal{C}_f$ en $\pm\infty$.
\end{tcolorbox}
\end{center}

\paragraph{Exemple 2 :} Soit $g(x) = \dfrac{2x^2 - 1}{x^2 + 3}$ définie sur $\mathbb{R}$.
\[
\lim_{x \to +\infty} g(x) = \lim_{x \to +\infty} \frac{2x^2}{x^2} = 2 \quad \text{et} \quad \lim_{x \to -\infty} g(x) = 2
\]
\textbf{Conclusion :} La droite d'équation $y = 2$ est une asymptote horizontale à la courbe $\mathcal{C}_g$ en $+\infty$ et en $-\infty$.

\subsubsection*{\color{myred}c. Asymptote oblique :}

\begin{center}
    \begin{tcolorbox}[colback=blue!5!white,colframe=myblue,title=\textbf{Définition : Asymptote oblique}]
La droite $(\mathcal{D}) : y = ax + b$ (avec $a \neq 0$) est une \textbf{asymptote oblique} à la courbe $\mathcal{C}_f$ en $\pm\infty$ si et seulement si :
\[
\lim_{x \to \pm\infty} [f(x) - (ax + b)] = 0
\]
\end{tcolorbox}
\end{center}

\paragraph{Exemple 3 :} Soit $h(x) = \dfrac{2x^2 - 3x + 5}{x - 1}$ définie sur $\mathbb{R} \setminus \{1\}$.
Montrons que la droite $(\mathcal{D}) : y = 2x - 1$ est asymptote oblique à $\mathcal{C}_h$ en $+\infty$.
\begin{enumerate}
    \item Calculons la différence $h(x) - (2x - 1)$ :
    \[
    h(x) - (2x - 1) = \frac{2x^2 - 3x + 5}{x - 1} - \frac{(2x - 1)(x - 1)}{x - 1} = \frac{(2x^2 - 3x + 5) - (2x^2 - 3x + 1)}{x - 1} = \frac{4}{x - 1}
    \]
    \item Calculons la limite de la différence :
    \[
    \lim_{x \to +\infty} [h(x) - (2x - 1)] = \lim_{x \to +\infty} \frac{4}{x - 1} = 0
    \]
\end{enumerate}
\textbf{Conclusion :} La droite d'équation $y = 2x - 1$ est une asymptote oblique à $\mathcal{C}_h$ en $+\infty$ (et de même en $-\infty$).

\subsection*{\color{myred}2. Branches paraboliques}

Lorsque $\lim_{x \to \pm\infty} f(x) = \pm\infty$, la courbe ne possède pas d'asymptote horizontale. On étudie alors la limite du rapport $\dfrac{f(x)}{x}$ :

\begin{tcolorbox}[colback=gray!5!white,colframe=black,title=\textbf{Définitions : Les deux directions paraboliques}]
\begin{itemize}
    \item \textbf{Branche parabolique de direction $(Oy)$ :}
    Si $\lim_{x \to \pm\infty} \frac{f(x)}{x} = \pm\infty$, la courbe $\mathcal{C}_f$ admet une branche parabolique dont la direction asymptotique est l'axe des ordonnées $(Oy)$.
    \item \textbf{Branche parabolique de direction $(Ox)$ :}
    Si $\lim_{x \to \pm\infty} \frac{f(x)}{x} = 0$, la courbe $\mathcal{C}_f$ admet une branche parabolique dont la direction asymptotique est l'axe des abscisses $(Ox)$.
\end{itemize}
\end{tcolorbox}

\paragraph{Exemple 4 : Branche parabolique de direction $(Oy)$}
Soit $f(x) = x^2 - 3x$ définie sur $\mathbb{R}$.
\begin{enumerate}
    \item $\lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} x^2 = +\infty$.
    \item Calculons la limite de $\dfrac{f(x)}{x}$ :
    \[
    \lim_{x \to +\infty} \frac{f(x)}{x} = \lim_{x \to +\infty} \frac{x^2 - 3x}{x} = \lim_{x \to +\infty} (x - 3) = +\infty
    \]
\end{enumerate}
\textbf{Conclusion :} $\mathcal{C}_f$ admet une branche parabolique de direction $(Oy)$ en $+\infty$.

\paragraph{Exemple 5 : Branche parabolique de direction $(Ox)$}
Soit $g(x) = \sqrt{x}$ définie sur $[0 \,;\, +\infty[$.
\begin{enumerate}
    \item $\lim_{x \to +\infty} g(x) = +\infty$.
    \item Calculons la limite de $\dfrac{g(x)}{x}$ :
    \[
    \lim_{x \to +\infty} \frac{g(x)}{x} = \lim_{x \to +\infty} \frac{\sqrt{x}}{x} = \lim_{x \to +\infty} \frac{1}{\sqrt{x}} = 0
    \]
\end{enumerate}
\textbf{Conclusion :} $\mathcal{C}_g$ admet une branche parabolique de direction $(Ox)$ en $+\infty$.
