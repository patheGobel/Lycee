\section*{\color{myred}I. Généralités}

\subsection*{\color{myred}1. Activité introductive}

\begin{tcolorbox}[colback=blue!5!white,colframe=myblue,title=\textbf{Activité : Découverte de la notion de suite}]
On considère la liste de nombres réels suivants : $-10 \,;\, 5 \,;\, 3 \,;\, 4 \quad \text{et} \quad -8$.

\begin{enumerate}
    \item Ranger ces $5$ nombres dans l'ordre \textbf{décroissant}.
    \item En attribuant le **rang $0$** au plus grand nombre, donner le rang de chacun des autres nombres de la liste ordonnée.
\end{enumerate}
\end{tcolorbox}

\vspace{0.2cm}

\paragraph{\color{mygreen}Correction de l'activité :}
\begin{enumerate}
    \item L'ordre décroissant des $5$ nombres est : $5 \,;\, 4 \,;\, 3 \,;\, -8 \,;\, -10$.
    \item Attribution des rangs :
    \begin{itemize}[label=$\bullet$]
        \item Le rang de $5$ est : \textbf{0}
        \item Le rang de $4$ est : \textbf{1}
        \item Le rang de $3$ est : \textbf{2}
        \item Le rang de $-8$ est : \textbf{3}
        \item Le rang de $-10$ est : \textbf{4}
    \end{itemize}
\end{enumerate}

\vspace{0.3cm}


\begin{tcolorbox}[colback=yellow!5!white,colframe=myred,title=\textbf{Exploitation pédagogique}]
\begin{itemize}[label=$\Rightarrow$]
    \item La liste ordonnée $5 \,;\, 4 \,;\, 3 \,;\, -8 \,;\, -10$ est un exemple de **suite de nombres réels**.
    \item Si l'on nomme cette suite $u$, chaque nombre de la liste est un **terme** de la suite $u$, repéré de manière unique par son **rang** (ou indice).
    \item Pour noter un terme, on écrit le nom de la suite et on place son rang en indice :
    \[
    u_0 = 5 \quad;\quad u_1 = 4 \quad;\quad u_2 = 3 \quad;\quad u_3 = -8 \quad;\quad u_4 = -10
    \]
    \item De façon générale, le terme situé au rang $n$ est noté $\mathbf{u_n}$ (lu \textit{« u indice n »} ou \textit{« u de n »}).
\end{itemize}
\end{tcolorbox}

\subsection*{\color{myred}2. Définition}

\begin{tcolorbox}[colback=red!5!white,colframe=myred,title=\textbf{Définition}]
On appelle \textbf{suite numérique} toute fonction $u$ définie d'une partie $E$ de $\mathbb{N}$ (souvent $\mathbb{N}$ lui-même) vers $\mathbb{R}$.

\[
\begin{array}{cccc}
u : & \mathbb{N} & \to & \mathbb{R} \\
    & n & \mapsto & u_n
\end{array}
\]
Le réel $u_n$ est appelé \textbf{terme général} ou \textbf{terme d'indice $n$}. La suite est notée $(u_n)_{n \in \mathbb{N}}$ ou simplement $(u_n)$.
\end{tcolorbox}

\subsection*{\color{myred}3. Modes de définition d'une suite}

\subsubsection*{\color{myblue}3.1. Suite explicite}
Lorsqu'une suite $(u_n)$ est exprimée directement en fonction de $n$, on dit qu'elle est définie par une \textbf{formule explicite} : $u_n = f(n)$.

\paragraph{Exemple :}
Soit la suite $(u_n)$ définie par $u_n = n^2 - 5n + 3$.
\begin{itemize}[label=$\bullet$]
    \item $u_0 = 0^2 - 5(0) + 3 = 3$
    \item $u_1 = 1^2 - 5(1) + 3 = -1$
    \item $u_2 = 2^2 - 5(2) + 3 = -3$
    \item $u_{10} = 10^2 - 5(10) + 3 = 53$
    \item $u_{50} = 50^2 - 5(50) + 3 = 2253$
\end{itemize}

\subsubsection*{\color{myblue}3.2. Suite définie par récurrence}
Lorsqu'un terme de la suite se calcule à partir du terme précédent, la suite est définie par une \textbf{relation de récurrence} : $u_{n+1} = f(u_n)$ avec la donnée du premier terme.

\paragraph{Exemple 1 :}
Soit la suite $(u_n)$ définie par $u_0 = 2$ et $u_{n+1} = \dfrac{2u_n + 3}{u_n + 1}$.
Calculons les premiers termes :
\begin{itemize}[label=$\bullet$]
    \item $u_1 = \dfrac{2u_0 + 3}{u_0 + 1} = \dfrac{2(2) + 3}{2 + 1} = \dfrac{7}{3}$
    \item $u_2 = \dfrac{2u_1 + 3}{u_1 + 1} = \dfrac{2(7/3) + 3}{7/3 + 1} = \dfrac{23/3}{10/3} = \dfrac{23}{10}$
\end{itemize}

\paragraph{Exemple 2 :}
Soit la suite $(u_n)$ définie par $u_0 = -1$ et $u_{n+1} = 2u_n + 3$.
\begin{itemize}[label=$\bullet$]
    \item $u_1 = 2u_0 + 3 = 2(-1) + 3 = 1$
    \item $u_2 = 2u_1 + 3 = 2(1) + 3 = 5$
    \item $u_3 = 2u_2 + 3 = 2(5) + 3 = 13$
\end{itemize}
\textit{Remarque :} Pour calculer $u_{50}$ avec une suite récurrente, il faut calculer successivement tous les termes précédents.

\begin{tcolorbox}[colback=yellow!5!white,colframe=myred,title=\textbf{Exercice d'application}]
Dans chacun des cas suivants, calculer les $6$ premiers termes de la suite $(u_n)_{n \in \mathbb{N}}$ :
\begin{enumerate}
    \item $u_n = 7n^2 - 5n + 2$
    \item $u_n = \dfrac{3n - 5}{n + 2}$
\end{enumerate}
\end{tcolorbox}

\subsection*{\color{myred}4. Sens de variation d'une suite}

\begin{tcolorbox}[colback=blue!5!white,colframe=myblue,title=\textbf{Définitions}]
Une suite $(u_n)$ est dite :
\begin{itemize}
    \item \textbf{Croissante} si pour tout $n \in \mathbb{N}$, $u_{n+1} \ge u_n \iff u_{n+1} - u_n \ge 0$.
    \item \textbf{Décroissante} si pour tout $n \in \mathbb{N}$, $u_{n+1} \le u_n \iff u_{n+1} - u_n \le 0$.
    \item \textbf{Monotone} si elle est soit uniquement croissante, soit uniquement décroissante.
    \item \textbf{Constante} si pour tout $n \in \mathbb{N}$, $u_{n+1} = u_n$.
\end{itemize}
\end{tcolorbox}

\paragraph{Règle générale :} Pour étudier le sens de variation d'une suite $(u_n)$, on étudie le signe de la différence $u_{n+1} - u_n$.

\paragraph{Exemple :}
Soit $u_n = \dfrac{n+2}{2n+1}$. Étudions le signe de $u_{n+1} - u_n$ :
\[
u_{n+1} = \dfrac{(n+1)+2}{2(n+1)+1} = \dfrac{n+3}{2n+3}
\]
\[
u_{n+1} - u_n = \dfrac{n+3}{2n+3} - \dfrac{n+2}{2n+1} = \dfrac{(n+3)(2n+1) - (n+2)(2n+3)}{(2n+1)(2n+3)} = \dfrac{-3}{(2n+1)(2n+3)}
\]
Comme $n \ge 0$, le dénominateur est strictement positif et le numérateur est $-3 < 0$. 
Ainsi, $u_{n+1} - u_n < 0$ : la suite $(u_n)$ est \textbf{strictement décroissante}.

\section*{\color{myred}II. Suites Arithmétiques}

\begin{tcolorbox}[colback=red!5!white,colframe=myred,title=\textbf{Définition}]
Une suite $(u_n)$ est \textbf{arithmétique} s'il existe un nombre réel $r$, appelé \textbf{raison}, tel que pour tout $n \in \mathbb{N}$ :
\[
u_{n+1} = u_n + r
\]
\end{tcolorbox}

\subsection*{\color{myred}1. Expression du terme général}
\begin{itemize}[label=$\bullet$]
    \item Si le premier terme est $u_0$ : $\mathbf{u_n = u_0 + nr}$
    \item Si le premier terme est $u_1$ : $\mathbf{u_n = u_1 + (n-1)r}$
    \item De manière générale, pour tout premier terme $u_p$ :
    \[
    \mathbf{u_n = u_p + (n-p)r}
    \]
\end{itemize}

\begin{tcolorbox}[colback=green!5!white,colframe=mygreen,title=\textbf{Méthode : Démontrer la nature d'une suite}]
\begin{itemize}
    \item **Pour prouver qu'une suite est arithmétique :** On calcule la différence $u_{n+1} - u_n$. Si le résultat est une constante $r$ (indépendante de $n$), alors la suite est arithmétique de raison $r$.
    \item **Pour prouver qu'une suite N'EST PAS arithmétique :** Il suffit de vérifier que $u_1 - u_0 \neq u_2 - u_1$.
\end{itemize}
\end{tcolorbox}
\subsection*{\underline{\textbf{\textcolor{red}{Exemple :}}}}
On considère la suite $(U_n)$ définie par :$ u_n = 5n + 3 $

Montrer que $(U_n)$ est une suite arithmétique dont on précisera la raison et le premier terme.

\subsection*{\underline{\textbf{\textcolor{red}{Contre Exemple :}}}}
Soit la suite $(U_n)$ définie par : $U_{n}=\frac{n}{n+1}$

Montrer que $(U_n)$ n’est pas une suite arithmétique.

\subsubsection*{\underline{\textbf{\textcolor{red}{Exercie :}}}}
Dans chacun des cas suivants, donner le terme général de la suite.
\begin{itemize}
\item[1.]$(u_n)$ est une suite arithmétique de raison $r = 3$ et de premier terme $u_{0}= 7$.
\item[2.]$(u_n)$ est une suite arithmétique de raison $r=-\frac{3}{4}$ et de premier terme 
$u_{1}=\frac{-7}{4}$.
\item[3.] $(U_n)$ est une suite arithmétique de raison $r$ et de premier terme $U_{0}$ tels que $U_{50} = 406$ et $U_{100} = 806$.
\end{itemize}

\subsubsection*{\underline{\textbf{\textcolor{red}{Exercie d'application  :}}}}

Dans une ville, le prix du transport augmente de $100\text{ F}$ chaque année. En $2019$, le prix était de $600\text{ FCFA}$.
On pose $u_0 = 600$ (prix en $2019$) et $u_n$ le prix en $2019 + n$.
\begin{enumerate}
    \item Calculer le prix du transport en $2020$ ($u_1$), $2021$ ($u_2$) et $2022$ ($u_3$).
    \item Exprimer $u_{n+1}$ en fonction de $u_n$.
    \item En déduire la nature de la suite $(u_n)$ et exprimer $u_n$ en fonction de $n$.
\end{enumerate}

\subsection*{\color{myred}2. Somme des termes consécutifs}
Pour une suite arithmétique, la somme $S_n = u_p + u_{p+1} + \dots + u_n$ est donnée par la formule :
\begin{tcolorbox}[colback=red!5!white,colframe=myred]
\[
S_n = \text{(Nombre de termes)} \times \dfrac{\text{Premier terme} + \text{Dernier terme}}{2} = \mathbf{\dfrac{(n - p + 1)(u_p + u_n)}{2}}
\]
\end{tcolorbox}

\subsection*{\underline{\textbf{\textcolor{red}{Exercie d'application :}}}}
Soit $\left(u_{n}\right)$ la suite définie par : $u_{n}=2n+7$
\begin{itemize}
\item[1.]Calculer $u_{1}$, $u_{2}$ et $u_{2024}.$
\item[2.] Montrer que (un) est une suite arithmétique.
\item[3.] Calculer la somme : $S_{n} = u_{0} + u_{1} + \cdots+ u_{n}.$
\item[4.] En déduire le somme $S = u_{1} + u_{2} +\cdots+ u_{2024}.$
\end{itemize}

\section*{\color{myred}III. Suites Géométriques}

\begin{tcolorbox}[colback=red!5!white,colframe=myred,title=\textbf{Définition}]
Une suite $(u_n)$ est \textbf{géométrique} s'il existe un nombre réel non nul $q$, appelé \textbf{raison}, tel que pour tout $n \in \mathbb{N}$ :
\[
u_{n+1} = u_n \times q
\]
\end{tcolorbox}

\subsection*{\color{myred}1. Expression du terme général}
\begin{itemize}[label=$\bullet$]
    \item Si le premier terme est $u_0$ : $\mathbf{u_n = u_0 \times q^n}$
    \item Si le premier terme est $u_1$ : $\mathbf{u_n = u_1 \times q^{n-1}}$
    \item De manière générale, pour tout premier terme $u_p$ :
    \[
    \mathbf{u_n = u_p \times q^{n-p}}
    \]
\end{itemize}

\begin{tcolorbox}[colback=green!5!white,colframe=mygreen,title=\textbf{Méthode : Démontrer la nature d'une suite}]
\begin{itemize}
    \item \textbf{Pour prouver qu'une suite est géométrique :} Pour tout $u_n \neq 0$, on calcule le quotient $\dfrac{u_{n+1}}{u_n}$. Si le résultat est une constante $q$ (indépendante de $n$), alors la suite est géométrique de raison $q$.
    \item \textbf{Pour prouver qu'une suite N'EST PAS géométrique :} Il suffit de vérifier que $\dfrac{u_1}{u_0} \neq \dfrac{u_2}{u_1}$.
\end{itemize}
\end{tcolorbox}

\subsection*{\underline{\textbf{\textcolor{red}{Exemple :}}}}
On considère la suite $(u_n)$ définie pour tout $n \in \mathbb{N}$ par : $u_n = 3 \times 2^n$.

Montrer que $(u_n)$ est une suite géométrique dont on précisera la raison $q$ et le premier terme $u_0$.

\subsection*{\underline{\textbf{\textcolor{red}{Contre-Exemple :}}}}
Soit la suite $(u_n)$ définie pour tout $n \in \mathbb{N}$ par : $u_n = n^2 + 1$.

Montrer que $(u_n)$ n'est pas une suite géométrique.

\subsubsection*{\underline{\textbf{\textcolor{red}{Exercice :}}}}
Dans chacun des cas suivants, donner le terme général de la suite.
\begin{itemize}
\item[1.] $(u_n)$ est une suite géométrique de raison $q = 3$ et de premier terme $u_0 = 5$.
\item[2.] $(u_n)$ est une suite géométrique de raison $q = \dfrac{1}{2}$ et de premier terme $u_1 = 8$.
\item[3.] $(u_n)$ est une suite géométrique à termes positifs, de raison $q$ et de premier terme $u_0$, tels que $u_2 = 12$ et $u_4 = 48$.
\end{itemize}

\subsubsection*{\underline{\textbf{\textcolor{red}{Exercice d'application 1 :}}}}
Une population de bactéries compte $1\,000$ individus au début de l'expérience (en $2024$). Chaque année, cette population augmente de $10\%$. On note $u_0 = 1\,000$ la population initiale et $u_n$ la population en $2024 + n$.
\begin{enumerate}
    \item Calculer la population en $2025$ ($u_1$), $2026$ ($u_2$) et $2027$ ($u_3$).
    \item Exprimer $u_{n+1}$ en fonction de $u_n$.
    \item En déduire la nature de la suite $(u_n)$ et exprimer $u_n$ en fonction de $n$.
\end{enumerate}

\subsection*{\color{myred}2. Somme des termes consécutifs ($q \neq 1$)}
Pour une suite géométrique de raison $q \neq 1$, la somme $S_n = u_p + u_{p+1} + \dots + u_n$ vaut :
\begin{tcolorbox}[colback=red!5!white,colframe=myred]
\[
S_n = \text{Premier terme} \times \dfrac{1 - q^{\text{Nombre de termes}}}{1 - q} = \mathbf{u_p \times \dfrac{1 - q^{n - p + 1}}{1 - q}}
\]
\end{tcolorbox}

\subsection*{\underline{\textbf{\textcolor{red}{Exercice d'application 2 :}}}}
Soit $(u_n)$ la suite géométrique de premier terme $u_0 = 3$ et de raison $q = 2$.
\begin{enumerate}
    \item Calculer $u_1$, $u_2$ et $u_{10}$.
    \item Exprimer $u_n$ en fonction de $n$.
    \item Calculer la somme : $S_n = u_0 + u_1 + \dots + u_n$.
    \item En déduire la somme $S = u_1 + u_2 + \dots + u_{10}$.
\end{enumerate}
